# འགོ་བཙུགས Iroha 3 {#launch-iroha-3}

སྒེར་གྱི་མཐུད་སྦྱོར་བརྟག་དཔྱད་འབད་ནི་ལུ་ Linux VM གཅིག་གུ་ Iroha 3 གི་བདེན་དཔྱད་འབད་མི་བཞི་གཡོག་བཀོལ། མིང་རྟགས་བཀོད་ཡོད་པའི་སྤྲོད་ལེན་ཅིག་གཏང་ཞིནམ་ལས་ བདེན་དཔྱད་འབད་མི་གཅིག་བཀག་བཞག ལྷག་ལུས་གསུམ་གྱིས་སྤྲོད་ལེན་གཞན་མི་ཅིག་སླར་ཡང་ངེས་གཏན་བཟོ་ཚུགས་ག་བལྟ། འདི་གིས་གནང་བ་ཡོད་མི་ཚུ་རྐྱངམ་ཅིག་འཛུལ་ཚུགས་པའི་རྩིས་དེབ་རང་དབང་ཅན་ཅིག་བཟོཝ་ཨིན། རྩིས་དེབ་འདི་སྐྱོང་སྲིད་ལྡན་པའི་གནས་སྡུད་ས་སྟོང་ཅིག་སྦེ་ SORA Nexus ལུ་མཐུད་ནི་ལུ་གྲ་སྒྲིག་སོ་སོ་འབད་དགོ།

## གྲ་སྒྲིག་འབད་ནི། VM {#prepare-the-vm}

Linux x86_64 གུ་ Docker Engine དང་ Buildx, Compose v2, Python 3.8 ཡང་ན་དེ་ལས་གསར་མི་ དེ་ལས་ `unzip` ལག་ལེན་འཐབ། ཁྱོད་ཀྱི་རྩིས་ཐོ་ལུ་ Docker ལག་ལེན་འཐབ་ནིའི་གནང་བ་དགོ། འགོ་དང་པ་བཟོ་བསྐྲུན་འབད་བའི་སྐབས་ GitHub, Docker Hub, Debian གྱི་ཐུམ་སྒྲིལ་གྱི་ངོ་བཤུས་སྦྱིན་མི་སར་བར་དང་ crates.io ལུ་ HTTPS ཐོག་ལས་མཐུད་ཚུགས་དགོ།

ང་བཅས་ཀྱིས་ vCPU དྲུག་དང་ RAM 32 GiB ཡོད་པའི་ VM གུ་ ལག་ཆས་འདི་བཟོ་ཞིནམ་ལས་ བརྟག་དཔྱད་ཀྱི་སྤྲོད་ལེན་དག་པ་ཅིག་གིས་ཚོད་བལྟ་འབད་ཡི། གསོག་སྟོར་གྱི་ཚད་འདི་ ལཱ་གི་འབོར་ཚད་དང་གནས་སྡུད་བཞག་ནིའི་དུས་ཡུན་ལུ་བལྟ་སྟེ་ཐག་བཅད། Docker Engine རེ་ལུ་ལག་ཆས་ཚར་རེ་ལག་ལེན་འཐབ་སྟེ་ loopback འདྲེན་ལམ་ 18080–18083 སྟོངམ་བཞག།

[བདེན་དཔྱད་འབད་མི་བཞི་གི་ལག་ཆས་ཕབ་ལེན་འབད](/downloads/iroha-trade-poc.zip)། འདི་ནང་འགོ་བཙུགས་མིའི་ལས་རིམ་དང་ ཐོན་རིམ་ངེས་གཏན་བཟོ་ཡོད་པའི་རྟེན་ཆའི་ཐོ་ཡིག་ དེ་ལས་ Iroha གི་ཐོན་རིམ་གཅིག་ལས་བདེན་དཔྱད་འབད་མི་དང་ཞབས་ཏོག་ལེན་མི་གཉིས་ཆ་རང་བཟོ་ནིའི་བཅོས་ཁ་ཚུ་ཡོད། ཕབ་ལེན་འབད་སའི་སྣོད་ཐོ་ནང་ཡིག་སྣོད་ཀྱི་ཞིབ་དཔྱད་གནས་གོང་བལྟ་ཞིནམ་ལས་ཐུམ་ཁ་ཕྱེ།

## བདེན་དཔྱད་འབད་མི་བཞི་ཆ་མཉམ་འགོ་བཙུགས་ནི་དང་བདེན་དཔྱད་འབད། {#start-and-verify-all-four-validators}

```sh
python3 poc.py up
python3 poc.py verify
```

འགོ་དང་པ་གཡོག་བཀོལ་བའི་སྐབས་ ལས་རིམ་གྱིས་སྣོད་ཀྱི་གཟུགས་བརྙན་བཟོ་སྟེ་ VM གུ་རྒྱུད་རིམ་གྱི་ངོ་རྟགས་དང་ལྡེ་མིག་ དེ་ལས་མིང་རྟགས་བཀོད་ཡོད་པའི་འགོ་དང་པའི་བཀག་ཆ་བཟོཝ་ཨིན། བདེན་དཔྱད་འབད་མི་རེ་ལུ་རང་གི་རྟག་བརྟན་གསོག་སྟོར་ཡོད། ཞིབ་དཔྱད་ལས་རིམ་གྱིས་བརྟག་དཔྱད་སྤྲོད་ལེན་ཅིག་གུ་མིང་རྟགས་བཀོད་དེ་ བདེན་དཔྱད་འབད་མི་བཞི་ཆ་རང་གིས་མཉམ་རོགས་གསུམ་དང་མཐུད་ཡོདཔ་དང་ བཀག་ཆ་གསརཔ་གི་ཧེཤ་གནས་གོང་གཅིག་འབད་ཡོདཔ་སྦེ་སྙན་ཞུ་འབད་ཚུན་སྒུགཔ་ཨིན། འགོ་དང་པའི་བཀག་ཆ་ནང་དཔེ་ཚད་རྩིས་ཐོ་ཚུ་ཡོད། བརྟག་དཔྱད་ཀྱི་གནས་སྡུད་ལག་ལེན་འཐབ་སྟེ་ APIs ཚུ་ loopback གུ་རྐྱངམ་ཅིག་བཞག།

ལས་ བདེན་དཔྱད་འབད་མི་དང་པ་གི་གནས་རིམ་ལྷག་ནི་ལུ་ VM, ཞུ༌བ༌ JSON གསལ་པོ།

```sh
curl -fsS -H 'Accept: application/json' http://127.0.0.1:18080/status
```

## བདེན་དཔྱད་འབད་མི་འཐུས་ཤོར་ཅིག་བརྟག་དཔྱད་འབད། {#test-a-validator-failure}

```sh
python3 poc.py failover
```

བརྟག་དཔྱད་འདི་གིས་བདེན་དཔྱད་འབད་མི་གཅིག་བཀག་སྟེ་ སྤྲོད་ལེན་གསརཔ་ཅིག་བཏང་ཞིནམ་ལས་ ལྷག་ལུས་གསུམ་གུ་ངེས་གཏན་བཟོ་ཡོད་པའི་བཀག་ཆའི་ཧེཤ་གནས་གོང་བལྟཝ་ཨིན། དེ་ལས་བཞི་པ་ལོག་འགོ་བཙུགས་ཏེ་ བཀག་ཆ་གཅིག་པ་ལུ་ལྷོད་ཚུན་སྒུགཔ་ཨིན། VM མེདཔ་འགྱོ་བ་ཅིན་ བདེན་དཔྱད་འབད་མི་བཞི་ཆ་རང་འགགཔ་ཨིན། དངོས་གཟུགས་ཀྱི་ཧོསཊི་གཅིག་མེདཔ་འགྱོ་བའི་ཚོད་བལྟ་འབད་ནི་ལུ་ རང་དབང་ཅན་གྱི་ཧོསཊི་བཞི་གུ་བདེན་དཔྱད་འབད་མི་རེ་རེ་བཞག།

## བཀག་ཞིནམ་ལས་ ལོག་འགོ་བཙུགས། {#stop-and-resume}

```sh
python3 poc.py down
python3 poc.py up
```

བརྡ་བཀོད་གཉིས་ཆ་རང་ ལག་ཆས་ཐུམ་ཁ་ཕྱེ་ཡོད་པའི་སྣོད་ཐོ་གཅིག་པ་ནང་ལས་གཡོག་བཀོལ། ལོག་འགོ་བཙུགསཔ་ད་ རྩིས་དེབ་གཅིག་པ་ཁ་ཕྱེ་ཚུགས་ནིའི་དོན་ལུ་ `state/` སྣོད་ཐོ་དང་ Docker གི་གསོག་སྟོར་བཞི་གཅིག་ཁར་བཞག། འགོ་བཙུགས་མ་ཚུགས་པ་ཅིན་ སྣོད་ཐོ་དེ་ནང་ `python3 poc.py logs` གཡོག་བཀོལ།

## མི་མང་ལུ་ཚོད་བགམ་འབད། API མཐའ་མཚམས། {#try-the-public-endpoint}

ལས་རིམ་ཅིག་གིས་ རྩིས་དེབ་ཀྱི་ VM མ་གཡོག་བཀོལ་བར་ Taira གུ་ཡོད་པའི་ SORA Nexus གི་ཡོངས་ཁྱབ་གནས་སྡུད་ས་སྟོང་ལས་ལྷག་ཚུགས། `curl` དང་ `jq` ཡོད་པའི་ལཱ་གཡོག་འཕྲུལ་ཆས་ལས་ བརྟག་དཔྱད་ཡོངས་འབྲེལ་ལུ་ལྷག་ནི་ཙམ་གྱི་ཞུ་བ་འདི་གཏང་།

```sh
curl -fsS -H 'Accept: application/json' https://taira.sora.org/status \
  | jq '{peers, blocks, txs_approved, queue_size}'
```

ལན་ནང་ ཞུ་བ་ཐོབ་མི་མཐུད་ས་དང་མཐུད་ཡོད་པའི་མཉམ་རོགས་ཚུ་གི་གྱངས་ཁ་ ངེས་གཏན་བཟོ་ཡོད་པའི་བཀག་ཆ་ཚུ་གི་གྱངས་ཁ་ ཆ་འཇོག་གྲུབ་པའི་སྤྲོད་ལེན་དང་གྱལ་རིམ་ནང་སྒུག་མི་སྤྲོད་ལེན་ཚུ་གི་གྱངས་ཁ་སྟོནམ་ཨིན། བཀྲམ་སྤེལ་འཆར་གཞིའི་དོན་ལུ་ [ཡོངས་འབྲེལ་བཀྲམ་སྤེལ](/dz/guide/best-practices/network-deployment.md) ལུ་གཟིགས།
