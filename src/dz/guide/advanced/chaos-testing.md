# Izanami དང་གཅིག་ཁར་ ཐད་ཀར་དུ་ བརྟག་དཔྱད་འབདཝ་ཨིན། {#chaos-testing-with-izanami}

Izanami འདི་ ཡར་རྒྱུད་ Iroha ལཱ་གི་ས་སྒོ་ནང་ ཅའོ་སི་ནེཊི་ཨོར་ཀེ་སི་ཊར་ཊར་ཨིན། འདི་གིས་ བཀོ་བཞག་བཏུབ་པའི་ ཉེ་གནས་ Iroha ཀླད་ཀོར་འགོ་བཙུགས་ཏེ་ རིམ་སྒྲིག་འབད་བཏུབ་པའི་ ལཱ་གི་མངོན་གསལ་ཅིག་ བཙུགས་ཏེ་ སེལ་འཐུ་འབད་ཡོད་པའི་ཡོངས་འབྲེལ་མཉམ་རོགས་ཚུ་ནང་ འཛོལ་བ་ཚུ་བཙུགསཔ་ལས་ བཀོལ་སྤྱོད་པ་ཚུ་གིས་ ཡོངས་འབྲེལ་འདི་ ཚད་འཛིན་འབད་ཡོད་པའི་འཐུས་ཤོར་གྱི་འོག་ལུ་ ཡར་རྒྱས་འགྱོ་དོ་ཡོདཔ་ཨིན་ན་མེན་ན་ བརྟག་ཞིབ་འབད་ཚུགས།

བཟོ་སྐྲུན་གྱི་ཧེ་མར་ བརྟག་ཞིབ་འབད་ནིའི་དོན་ལུ་ Izanami ལག་ལེན་འཐབ་། སླར་ལོག་སླར་གསོ་འབད་ནི་དང་ མཉམ་མཐུན་བཟོ་ཐབས། གློག་ཆས་འདི་ བཟོ་སྐྲན་གི་དྲ་ལམ་ལུ་མ་བཏོན་ ག་ཅི་སྨོ་ཟེར་བ་ཅིན་ གློག་ཆས་འདི་གིས་ འགོ་འདྲེན་འཐབ་མི་ འདྲ་མཉམ་ཚུ་རྩིས་ཏེ་ འདྲ་མཉམ་བསྐྱར་འགོ་བཙུགསཔ་ད། སྒྲིང་སྒྲི་སྣུམ་ དེ་ལས་ གནས་སྐབས་ཀྱི་ བློ་གཏད་ཅན་གྱི་ འདྲ་མཉམ་ཆ་བགོས་ དེ་ལས་ ས་གནས་ནང་ CPU ཡང་ན་ དཔར་སི་ཨེབ་གཏང་འབད་ནིའི་ ཐབས་ཤེས་ཡོད།

## དགོས་མཁོ་ཚུ་ {#prerequisites}

Izanami འདི་ ལཱ་འབད ལས་ [Iroha གཞི་རྟེན་རྩིས་སྲུང་ཁང་](https://github.com/hyperledger-iroha/iroha), གནད་སྡུད་ཡིག་ཚང་འདི་ནང་ལས་མེན་:

```bash
git clone https://github.com/hyperledger-iroha/iroha.git
cd iroha
cargo build -p izanami
```

གློག་ཐིག་གཉིས་ཆ་ར་ལུ་ འབྲེལ་མཐུད་འབད་ཡོད་པའི་མཐུད་མཚམས་ཚུ་ བཟོ་ནི་དང་ ལག་ལེན་འབད་ནི་ལུ་ ཐབས་ཤེས་གསལ་སྦེ་བྱིན་དགོཔ་ཨིན། TUI ལས་མ་འོང་མི་ བཀྲམ་སྤེལ་ཐེངས་རེ་གི་དོན་ལུ་ `--allow-net` མཐར་འཁྱོལ་ ཡང་ན་ TUI ནང་ `allow_net` སྒྲིག་འཇུག་འདི་བཟོ་བཅད།

```bash
cargo run -p izanami -- --allow-net --peers 4 --faulty 1 --duration 120s
```

འབྲེལ་མཐུད་ཅན་གྱི་གཡོག་བཀོལ་རིམ་སྒྲིག་ཅིག་གི་དོན་ལུ་:

```bash
cargo run -p izanami -- --tui --allow-net
```

Izanami གིས་ ལག་ལེན་པའི་རིམ་སྒྲིག་སྣོད་ཐོ་འོག་ལུ་ TUI དང་ CLI སྒྲིག་སྟངས་ཚུ་ གནས་ཏེ་ཡོདཔ་ཨིན། འགོ་དང་པ་ཐོན་པའི་ཡིག་སྣོད་ལུ་ གསལ་ཏོག་ཏོ་ V1 བཀོད་སྒྲིག་བཱའིཊི་གཅིག་ཡོདཔ་ཨིན། སྔོན་འགྲོའི་གསར་བཏོན་ཡང་ན་ དེ་མེན་པ་ཅིན་ ཐོན་རིམ་མ་འབད་བའི་སྒྲིག་སྟངས་ཚུ་ ངོས་ལེན་མ་འབད་བར་ཡོདཔ་ལས་ གནས་སྤོ་འབད་ནི་ལས་ ལོག་གསར་བསྐྲུན་འབད་དགོ། ད་ལྟོའི་གསལ་སྡུད་ཅིག་ལོག་ལག་ལེན་མ་འཐབ་པའི་ཧེ་མ་ བཀྲམ་སྟོན་འབད་ཡོད་པའི་སྒྲིག་སྟངས་ཚུ་བསྐྱར་ཞིབ་འབད།

## གཞི་རྟེན་ལམ་སྟོན་ {#baseline-run}

འཛོལ་བ་ཚབས་ཆེན་ཚུ་ཁ་སྐོང་མ་འབད་བའི་ཧེ་མ་ བསྐྱར་བཟོ་འབད་བཏུབ་པའི་གཞི་རྟེན་གཅིག་ལས་འགོ་བཙུགས།

```bash
cargo run -p izanami -- \
  --allow-net \
  --peers 4 \
  --faulty 1 \
  --duration 5m \
  --target-blocks 100 \
  --progress-interval 15s \
  --progress-timeout 120s \
  --latency-p95-threshold 2s \
  --tps 15 \
  --max-inflight 32 \
  --submitters 1 \
  --seed 42
```

འ་ནི་གཡོག་བཀོལ་འདི་ ཀླད་ཀོར་འདི་གིས་ ཞུ་བ་འབད་ཡོད་པའི་སྡེབ་ཚན་དམིགས་གཏད་ལུ་ལྷོད་པ་ཅིན་ དུས་ཚོད་མཇུག་བསྡུ་བའི་ནང་འཁོད་ལུ་ ཡར་རྒྱས་འགྱོ་སྟེ་སྡོད་པ་ཅིན་ དེ་ལས་ གདམ་ཁ་ཅན་གྱི་ p95 སྡེབ་ཚན་བར་མཚམས་ཚད་གཞི་འོག་ལུ་སྡོད་པ་ཅིན་རྐྱངམ་ཅིག་ མཐར་འཁྱོལ་འབྱུང་འོང་།

བརྡ་བཀོད་དང་ སོན་ Iroha ཁས་ལེན་ ཡོངས་འབྲེལ་མཉམ་རོགས་གྱངས་ཁ་ འཛོལ་བ་-མཉམ་རོགས་གྱངས་ཁ་ ལཱ་གི་མངོན་གསལ་གསལ་སྡུད་ དམིགས་གཏད་ TPS དེ་ལས་ དྲན་ཐོ་ཚུ་དང་གཅིག་ཁར་ འགོར་ཚད་ཚད་གཞི་ཚུ་ཐོ་བཀོད་འབད། གནས་གོང་འདི་ཚུ་མེད་པ་ཅིན་ གཞན་མི་བཀོལ་སྤྱོད་པ་ཅིག་གིས་ འཐུས་ཤོར་གྱི་དཔེ་གཞི་གཅིག་པ་འདི་ ལོག་གཏང་མི་ཚུགས།

## ལཱ་གི་འགན་ཁུར་གྱི་ཐོ་ཡིག་ཚུ་ {#workload-profiles}

ཨི་ཛ་ན་མི་ལུ་ལཱ་འགན་གསལ་སྡུད་གཉིས་ཡོདཔ་ཨིན།

| གསལ་སྡུད། | འདི་གི་དོན་ལུ་ལག་ལེན་འཐབ། | དྲན་ཐོ། |
| -------- | | |
| `stable` | ཡུན་རིངམོ་སྦེ་བཏང་ནི་དང་ བསྐྱར་བཟོ་འབད་བཏུབ་པའི་ ལཱ་འགན་ཞིབ་དཔྱད་ཚུ་ | ལག་བསྟར་ཉེན་སྲུང་ཅན་གྱི་བཟོ་ཐངས་ཚུ་ལུ་ཕན་ཐོགས། |
| `chaos` | འཐུས་ཤོར་-འགྲུལ་ལམ་ཁྱབ་ཁོངས། | བསམ་བཞིན་དུ་ནུས་མེད་བཟོ་ཐབས་ཚུ་ཚུདཔ་ཨིན། |

དང་པ་ བརྟན་ཏོག་ཏོ་གསལ་སྡུད་ལག་ལེན་འཐབ།

```bash
cargo run -p izanami -- --allow-net --workload-profile stable --seed 42
```

གཞི་རྟེན་འདི་ཧེ་མ་ལས་ཧ་གོ་ཚར་བའི་སྐབས་ ཟང་ཟིང་གསལ་སྡུད་ལུ་སོར་བསྒྱུར་འབད།

```bash
cargo run -p izanami -- --allow-net --workload-profile chaos --seed 42
```

གན་རྒྱ་བཀྲམ་སྤེལ་གྱི་ཐབས་ཤེས་ཚུ་ གསལ་ཏོག་ཏོ་སྦེ་ གནང་བ་མ་བྱིན་ཚུན་ཚོད་ བརྟན་ཏོག་ཏོ་གཡོག་བཀོལ་ཚུ་ནང་ ལྕོགས་མིན་བཟོཝ་ཨིན།

```bash
cargo run -p izanami -- \
  --allow-net \
  --workload-profile stable \
  --allow-contract-deploy-in-stable
```

གཡོག་བཀོལ་མི་འདི་གིས་ ཡར་རྒྱུན་ལཱ་གི་ས་སྒོ་ལས་ བཙུགས་ཡོད་པའི་ SORA Nexus སྔོན་སྒྲིག་ཚུ་ལག་ལེན་འཐབ་དགོཔ་ད་ `--nexus` ལག་ལེན་འཐབ།

## འཛོལ་བ་སེལ་འཐུ་འབད་ནི་ {#fault-controls}

`--faulty` འདི་ ཀླད་ཀོར་ལས་སྦོམ་པའི་སྐབས་ ཉུང་མཐའ་ལུ་ འཛོལ་བའི་གནས་སྟངས་གཅིག་ ལྕོགས་ཅན་བཟོ་དགོ། འཛོལ་བ་གིས་ སྔོན་སྒྲིག་འདི་ལྕོགས་ཅན་བཟོ་ཡོདཔ་ལུ་སོར་བསྒྱུར་འབདཝ་ཨིནམ་དང་ བུ་ལིན་དར་ཆ་ཚུ་ `=false` དང་ཅིག་ཁར་ལྕོགས་མིན་བཟོ་ཚུགས།

| འཛོལ་བ་ | CLI དར་ཆ | ག་ཅི་སྦྱོང་བརྡར་འབདཝ་ཨིན་ན། |
| ----------------------- | | |
| བརྡབ་འཁྲུག་དང་ལོག་འགོ་བཙུགས། | `--fault-enable-crash-restart` | མཉམ་རོགས་བྱ་རིམ་གྱི་གྱོང་གུན་དང་སླར་གསོ། |
| གསོག་འཇོག་བསུབ་ཞིནམ་ལས་ ལོག་འགོ་བཙུགས། | `--fault-enable-wipe-storage` | བརླག་སྟོར་ཤོར་བའི་ས་གནས་མངའ་སྡེ་ལས་ ལོག་ཐོབ་ཡོདཔ། |
| ནུས་མེད་ཚོང་འབྲེལ་སི་པམ་ | `--fault-enable-spam-invalid-transactions` | འཛུལ་ཞུགས་དང་ཕྱིར་འཐེན་གྱི་ལམ། |
| ཡོངས་འབྲེལ་གྱི་འགོར་འགྱངས་ | `--fault-enable-network-latency` | མགྱོགས་པའི་གཏམ་བཤད་དང་ཕྱིར་འགྱངས་ཀྱི་མོས་མཐུན་བརྡ་འཕྲིན། |
| ཡོངས་འབྲེལ་བར་བཅད་ | `--fault-enable-network-partition` | གནས་སྐབས་བློ་གཏད་ཅན་གྱི་མཉམ་རོགས་ཟུར་གནས་ |
| CPU གནོན་ཤུགས་ | `--fault-enable-cpu-stress` | ས་གནས་ཀྱི་བདེན་དཔྱད་དང་དུས་ཚོད་བཀོད་ནི་གི་གནོན་ཤུགས། |
| ཌིཀསི་ཚད་གཞི་ | `--fault-enable-disk-saturation` | ས་གནས་ཀྱི་གསོག་འཇོག་གནོན་ཤུགས། |

ཡོངས་འབྲེལ་-བར་བཅད་-རྐྱངམ་ཅིག་གཡོག་བཀོལ་ནིའི་དོན་ལུ་:


## གྲུབ་འབྲས་ཚུ་འགྲེལ་བཤད་འབདཝ་ཨིན། {#interpreting-results}

བཀྲམ་སྤེལ་འབད་ནིའི་ལཱ་འདི་ གསལ་ཏོག་ཏོ་ཅིག་སྦེ་ ལག་ལེན་འཐབ་དོ་ཡོདཔ་ད་ བཙག་འཐུ་འབད་ཡོད་མི་ མཐུད་མཚམས་མཉམ་པ་ཚུ་གིས་ ལྡེ་མིག་ཚུ་ བསྡུ་སྒྲིག་འབད་ནི་ལུ་ འཕྲོ་མཐུད་དེ་ར་ འབད་དོ་ཡོདཔ་དང་ ལྡེ་ཁུག་མེད་པའི་རྒྱབ་སྐྱོར་ཡར་སེང་འགྱོ་མི་ དེ་ལས་ གཞི་སྒྲིག་འབད་ཡོད་པའི་ སྒོ་སྒྲིག་མཇུག་བསྡུ་བའི་ཤུལ་ལུ་ ལྕོགས་གྲུབ་མེདཔ་ལས་ སླར་ལོག་འབད་ནི་གི་ ལཱ་འགོ་བཙུགས་ནི་མཚམས་འཇོག་འབདཝ་ཨིན།

གཡོག་བཀོལ་འདི་འཐུས་ཤོར་སྦེ་བརྩི།

- `--progress-timeout` ལས་རིངམོ་སྦེ་ ཡར་འཕེལ་གྱི་ས་ཁོངས་ཚུ་བཀག་བཞགཔ་ཨིན།
- མཉམ་རོགས་ཀྱི་མཐོ་ཚད་ཁ་ཕྱེ་སྟེ་ ལོག་སྟེ་མཉམ་འཛོམས་མི་འབད།

- ངོས་ལེན་མ་འབད་བར་ ཡང་ན་ དུས་ཚོད་རྫོགས་ཡོད་པའི་ཚོང་འབྲེལ་ཚུ་ སེལ་འཐུ་འབད་ཡོད་པའི་ལཱ་གི་མངོན་གསལ་གྱིས་ འགྲེལ་བཤད་མ་རྐྱབ་པས།
- མཉམ་རོགས་ལོག་འགོ་བཙུགས་ གསོག་འཇོག་འཕྱག་བདའ་ནི་ ཡང་ན་ བར་བཅད་སླར་གསོ་འབད་ནི་ལུ་ ལག་ཐོག་ལས་ གཙང་སྦྲ་འབད་དགོཔ་ཨིན།

འཐུས་ཤོར་བྱུང་པའི་ཤུལ་ལས་ སོན་ཅོག་འཐདཔ་དང་ འཛོལ་བ་དབྱེ་བ་གཅིག་ཉུང་སུ་ཅིག་དང་གཅིག་ཁར་ ལོག་གཡོག་བཀོལ། འདི་གིས་ འཐུས་ཤོར་གྱི་ཁ་ཐོག་འདི་ རྒྱ་ཆུང་ཀུ་བཟོ་བའི་སྐབས་ ལཱ་གི་འབོར་ཚད་དང་ དུས་ཚོད་འདི་ བསྐྱར་བཟོ་འབད་ཚུགསཔ་སྦེ་བཞགཔ་ཨིན།
