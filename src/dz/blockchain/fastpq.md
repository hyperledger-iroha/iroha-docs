མངའ་ཁོངས་-ཁ་ཕྱེ་ཡོད་པའི་ས་སྒོའི་ཧ་ཤི་ཚུ་ འདི་བཟུམ་སྦེ་ངོས་འཛིན་འབདཝ་ཨིན།

$$
H_D(m) =
H_F(
|\operatorname{pack}(D)|,\operatorname{pack}(D),
|\operatorname{pack}(m)|,\operatorname{pack}(m)
)
$$

བཱའིཊི་མངའ་ཁོངས་ཀིརིཔ་ཊོ་གཱ་ར་ཕིག་ཌའི་ཇེསཊི་ལས་འགོ་བཙུགས་མི་ ཀིརིཔ་ཊོ་གཱ་ར་ཕིག་ཧ་ཤི་ཚུ་གི་དོན་ལུ་ FastPQ བཱའིཊི་ཆུང་ཀུ་བརྒྱད་འགོ་ཐོག་འདི་ས་སྒོ་ནང་ལུ་སབ་ཁྲ་བཟོཝ་ཨིན།

$$
\operatorname{seed}(D)=
\operatorname{le64}(\operatorname{Hash}(D)[0..8])\bmod p
$$

འ་ནཱ་ `Hash` གིས་ Iroha གི་ `iroha_crypto::Hash::new` ཟེར་མི་འདི་ Blake2bVar གྱི་ 32-བཱའིཊི ཌི་ཇི་ཨེསི་ཨིནམ་མ་གཏོགས་ དཔེར་ན་ མན་ངངས ཅིག་གིས་ Poseidon2 ཡང་ན་ SHA-256 ལུ་ གསལ་ཏོག་ཏོ་སྦེ་མིང་བཏགས་མ་བྱིན་པར་ཨིན།

### ས་སྒོའི་ཨང་རྩིས་ {#field-arithmetic}

འདི་ཚུ་ Rust ལས་རིམ་ཨང་ཡིག གིས་ ཞིང་ ཆ་ཤས་གཙོ་བོཚུ་ ཚད་ལྡནསྦེ་ངོ་ཚབ་འབདཝ་ཨིན། `u64` གནས་གོང་ཚུ་ནང་ `[0,p)`. ཡར་སེང་དང་མར་ཕབ་ནི་ འདི་ཚུ་ཨིན།

$$
a +_F b = (a+b)\bmod p
$$

$$
a -_F b = (a-b)\bmod p
$$

བསྒྱུར་རྩིས་དང་པ་ ༡༢༨-བིཊི་ཐོན་འབྲས་རྩིས་སྟོནམ་ཨིན།

$$
a\cdot b = \operatorname{lo} + 2^{64}\operatorname{hi}
$$

གཱོལ་ཌི་ལོག་མར་ཕབ་འབད་ཞིནམ་ལས་ ངོ་རྟགས་འདི་ལག་ལེན་འཐབ་ཨིན།

$$
2^{64}\equiv2^{32}-1\pmod p
$$

གལ་ཏེ་:

$$
\operatorname{hi}=\operatorname{hi}_{lo}+2^{32}\operatorname{hi}_{hi}
$$

དེ་ལས་ མར་ཕབ་འབད་མི་གིས་ རྩིས་རྐྱབ་ཨིན།

$$
\operatorname{lo}
+2^{32}\operatorname{hi}_{lo}
-\operatorname{hi}_{lo}
-\operatorname{hi}_{hi}
\pmod p
$$

ལག་ལེན་འཐབ་མི་འདི་གིས་ གྲུབ་འབྲས་འདི་ ཚད་ལྡན་མ་འགྱོ་ཚུན་ཚོད་ `p` འདི་ གནས་སྟངས་དང་འཁྲིལ་ཏེ་ ཁ་སྐོང་ཡང་ན་ ཕབ་རྩིས་འབདཝ་ཨིན། མིང་རྟགས་བཀོད་ཡོད་པའི་ཧྲིལ་གྲངས་ཚུ་ དཔེར་ན་ ལྷག་ལུས་ཌེལ་ཊསི་ཚུ་ འདི་གིས་བཙུགས་ཡོདཔ་ཨིན།

$$
\operatorname{field}(x)=x\bmod p,\qquad 0\leq\operatorname{field}(x)<p
$$

### Poseidon2 བསྒྱུར་བཅོས་ {#poseidon2-permutation}

Poseidon2 བརྗེ་སྒྱུར་གནས་སྟངས་ནི།

$$
\mathbf{x}=(x_0,x_1,x_2)\in F^3
$$

དེའི་S-boxནི།

$$
S(x)=x^5
$$

FastPQ གིས་ སྐོར་ཐེངས་ཆ་ཚང་བཞི་དང་ ཆ་ཤས་སྐོར་ཐེངས་ལྔ་བཅུ་བདུན་ དེ་ལས་ སྐོར་ཐེངས་ཆ་ཚང་བཞི་ ད་རུང་ལག་ལེན་འཐབ་ཨིན། སྐོར་རིམ་ངེས་ཚིག་ `c_r = (c_{r,0}, c_{r,1}, c_{r,2})` དང་ཅིག་ཁར་ སྐོར་ཐེངས་ཆ་ཚང་ཅིག་འདི་:

$$
\mathbf{x}' =
M\cdot
\begin{bmatrix}
S(x_0+c_{r,0})\\
S(x_1+c_{r,1})\\
S(x_2+c_{r,2})
\end{bmatrix}
$$

ཆ་ཤས་སྐོར་ཐེངས་གཅིག་ནི།

$$
\mathbf{x}' =
M\cdot
\begin{bmatrix}
S(x_0+c_{r,0})\\
x_1+c_{r,1}\\
x_2+c_{r,2}
\end{bmatrix}
$$

ཁ་སྐོང་ཚུ དང་ བསྒྱུར་གྲངས་ཚུ ཆ་མཉམ་འདི་ `F`ནང་ལུ་ཡོདཔ་ཨིན། ཚད་ལྡན MDS གྲལ་སྒྲིག འདི་འདི་ཨིན།

$$
M=
\begin{bmatrix}
\texttt{0x982513a23d22b592} & \texttt{0xa3115db8cf1d9c90} & \texttt{0x46ba684b9eee84b7}\\
\texttt{0xbe3dce25491db768} & \texttt{0xfb0a6f731943519f} & \texttt{0xfce5bd953cde1896}\\
\texttt{0xe624719c41eb1a09} & \texttt{0xd2221b0f1aa2ebc4} & \texttt{0x1ab5e60d03ad44bc}
\end{bmatrix}
$$

ས་སྒོའི་ཧ་ཤི་འདི་ཀླད་ཀོར་གནས་སྟངས་ལས་འགོ་བཙུགསཔ་ཨིན། ཚད་གཞི་-༢ སྡེབ་ཚན་ཆ་ཚང་རེ་རེའི་དོན་ལུ་ `(u,v)`:

$$
(x_0,x_1,x_2)\leftarrow
\operatorname{Poseidon2}(x_0+u,x_1+v,x_2)
$$

མཐའ་མཇུག་སྡེབ་ཚན་འདི་གིས་ མཐའ་མཇུག་བརྗེ་སོར་གཅིག་གི་ཧེ་མ་ `1` བཀང་ཆ་ཆ་ཤས་འདི་ཁ་སྐོང་འབདཝ་ཨིན། ཐོན་འབྲས་འདི་ `x_0` ཨིན།

\qquad

\operatorname{slot}=\operatorname{saturating\_mul}

0^{32},& \text{if there are no permission hashes}\\

### ཨང་གྲངས་ཀྱི་ཚད་ལྡན་བཟོ་ནི་ {#numeric-normalization}

སྤོ་བཤུད་ཌེལ་ཊ་རེ་རེ་གི་དོན་ལུ་ དམིགས་གཏད་བཅུ་ཚག་འཇལ་ཚད་འདི་ དངུལ་ཚད་དང་ ལྷག་ལུས་པར་རིས་གཉིས་ཆ་རའི་ནང་ མཐོ་ཤོས་བཏོག་ཡོད་པའི་འཇལ་ཚད་ཨིན།

$$
s =
\max(
\operatorname{scale}(a),
\operatorname{scale}(f_0),
\operatorname{scale}(f_1),
\operatorname{scale}(t_0),
\operatorname{scale}(t_1)
)
$$

མན་ཊི་ས་ `Numeric` དང་ ཚད་གཞི་ `m` ཡོད་པའི་ `q` གནས་གོང་འདི་ `m >= 0` དང་ `q <= s` ཡོད་པའི་སྐབས་རྐྱངམ་ཅིག་ ངོས་ལེན་འབདཝ་ཨིན། དེ་གི་ FastPQ དཔང་པོའི་གནས་གོང་འདི་:

$$
\operatorname{norm}_s(m,q)=m\cdot10^{s-q}
$$

སྤྱིར་བཏང་བཟོ་ཡོད་པའི་གྲུབ་འབྲས་འདི་ `u64` ནང་ལུ་ཚུད་དགོ།

### ཀ་ནོ་སི་ཀཱན་གྱི་བཀའ་རྒྱ་ {#canonical-ordering}

འཚོལ་ཞིབ་བཟོ་བསྐྲུན་གྱི་ཧེ་མ་ བེཊི་འདི་ འགྱུར་བའི་ལྡེ་མིག་དང་ བཀོལ་སྤྱོད་གནས་རིམ་ དེ་ལས་ ངོ་མ་བཙུགས་ཟུར་ཐོ་ཚུ་གིས་ དབྱེ་སེལ་འབདཝ་ཨིན།

$$
r(\operatorname{Transfer})=0,\quad
r(\operatorname{Mint})=1,\quad
r(\operatorname{Burn})=2,\quad
r(\operatorname{RoleGrant})=3,\quad
r(\operatorname{RoleRevoke})=4,\quad
r(\operatorname{MetaSet})=5
$$

བཀའ་རྒྱ་བཀོད་པའི་འགན་ཁུར་འདི་ Poseidon2 ས་ཁོངས་ཀྱི་ཧེཤ་དེ་ མངའ་ཁོངས `fastpq:v1:ordering` དང་ Norito ཨེབ་གཏང་འབད་ཡོད་པའི་བརྒྱུད་རིམ་ཚུ་ཨིན།

$$
\operatorname{ordering\_hash} =
H_F(
|P(D_o)|,P(D_o),|P(E(T^\star))|,P(E(T^\star))
)
$$

འདི་ནང་ལུ་ `P`འདི་ བི་ཊ་༧ གི་ཆ་ཚན་ཨིནམ་ད་ `E`འདི་ Norito ཨེབ་གཏང་འབད་ནི་དང་ `D_o`དེ་ `fastpq:v1:ordering` དེ་ལས་ `T*`འདི་དབྱེ་སྒྲིག་འབད་ཡོད་པའི་འགྱུར་བཅོས་གི་ཐོ་ཡིགཨིན་མས།

### བརྗེ་སོར་གྱི་ཆ་སྙོམས་ཚུ་ {#transfer-equations}

སྐྱིན་འགྲུལ་བསྐྱོད་གྱི་དོན་ལུ་ `a`, བཀྲམ་སྤེལ་འབད་ཐངས་ `f`, དེ་ལས་ སྐྱིན་འགྲུལ་ལེན་མི་གི་མཐུན་རྐྱེན་ `t`, FastPQ མཐུན་རྐྱེན་ཚུ་བཟོ་མ་ཚར་བའི་ཧེ་མ་ རྟགས་མཚན་གྱི་གནས་གོང་ཚུ་ གཏན་འཁེལ་འབད་:

$$
f_0 \geq a
$$

$$
f_1 = f_0 - a
$$

$$
t_1 = t_0 + a
$$

འཕོ་སོར་གྲལ་ཐིག་ཚུ་གིས་ དེ་ལས་ཨིན་ཀོཌི་འབདཝ་ཨིན།

$$
\Delta_{\text{sender}} = f_1 - f_0 = -a
$$

$$

## བཟོ་བཀོད་ {#configuration}

FastPQ སྒྲིག་གཞི་འདི་ `zk.fastpq` གི་འོག་ལུ་བཞག་ཡོདཔ་ཨིན།

```toml
[zk.fastpq]
execution_mode = "cpu"
poseidon_mode = "cpu"

# Optional telemetry labels.
device_class = "apple-m4"
chip_family = "m4"
gpu_kind = "integrated"

# Optional Metal backend tuning.
metal_queue_fanout = 3
metal_queue_column_threshold = 24
metal_max_in_flight = 5
metal_threadgroup_width = 128
metal_trace = false
metal_debug_enum = false
metal_debug_fused = false
```

`iroha3d` ལས་ ལག་ལེན་འཐབ་ཐངས་དང་ ཊེ་ལི་མི་ཊི་རི་ ཁ་ཡིག་ཚུ་ བཀག་ཆ་འབད་ཚུགས།

```shell
iroha3d --fastpq-execution-mode gpu
iroha3d --fastpq-poseidon-mode cpu
iroha3d --fastpq-device-class apple-m4
iroha3d --fastpq-chip-family m4
iroha3d --fastpq-gpu-kind integrated
```

གནས་སྟངས་ཀྱི་འགྱུར་ལྡེ་ཚུ་ཡང་སྒྲིག་གཞི་གྱི་ས་ཁོངས་ཚུ་གི་དོན་ལུ་ རྒྱབ་སྐྱོར་འབད་ཡོདཔ་ཨིན། FastPQ གི་ཐད་ལུ་འགྱུར་ལྡེ་ཚུ་ཡང་:

- `FASTPQ_EXECUTION_MODE`
- `FASTPQ_POSEIDON_MODE`
- `FASTPQ_DEVICE_CLASS`
- `FASTPQ_CHIP_FAMILY`
- `FASTPQ_GPU_KIND`
- `FASTPQ_METAL_QUEUE_FANOUT`
- `FASTPQ_METAL_COLUMN_THRESHOLD`
- `FASTPQ_METAL_MAX_IN_FLIGHT`
- `FASTPQ_METAL_THREADGROUP`
- `FASTPQ_METAL_TRACE`
- `FASTPQ_DEBUG_METAL_ENUM`
- `FASTPQ_DEBUG_FUSED`
