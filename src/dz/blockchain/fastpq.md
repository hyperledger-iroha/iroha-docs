## རིམ་སྒྲིག་ {#configuration}

FastPQ རིམ་སྒྲིག་འདི་ `zk.fastpq` གི་འོག་ལུ་བཞག་ཡོདཔ་ཨིན།

```toml
[zk.fastpq]
execution_mode = "cpu"
poseidon_mode = "cpu"
proof_sidecar_queue_cap = 1024
proof_sidecar_max_retries = 16

# Optional telemetry labels.
device_class = "apple-m4"
chip_family = "m4"
gpu_kind = "integrated"

# Optional Metal backend tuning.
metal_queue_fanout = 3
metal_queue_column_threshold = 24
metal_max_in_flight = 5
metal_threadgroup_width = 128
metal_trace = false
metal_debug_enum = false

[zk.fastpq.producer]
poll_interval_ms = 1000
step_timeout_ms = 30000
max_native_steps = 16
max_source_entries = 65536
max_source_statements = 16384
```

ཐབས་ལམ་གཉིས་ཆ་ར་གིས་ `cpu` ཡང་ན་ `gpu` རྐྱངམ་ཅིག་ ངོས་ལེན་འབདཝ་ཨིནམ་དང་ སྔོན་སྒྲིག་འདི་ `cpu` ཨིན། དེ་ཚུ་ནང་ལས་ ག་འདི་ཨིན་རུང་ གཅིག་ `gpu` ལུ་ གཞི་སྒྲིག་འབད་བ་ཅིན་ བདེན་ཁུངས་བསྡུས་རྟགས་བཟོ་ནི་ལུ་ ཐབས་འཕྲུལ་ཅིག་ དགོཔ་ཨིན། macOS ནང་ Metal དང་ གཞན་ས་ཁར་ CUDA ཨིན། GPU རྒྱབ་སྐྱོར་འདི་ བཟོ་བསྐྲུན་ནང་ མ་བཙུགས་པ་ཅིན་ ཡང་ན་ ཐབས་འཕྲུལ་འདི་ སྔོན་ཞིབ་ནང་ འཐུས་ཤོར་བ་ཅིན་ བདེན་ཁུངས་བཟོ་མིའི་རྒྱུན་ལམ་འདི་ ལྕོགས་མིན་སྦེ་ར་ སྡོདཔ་ཨིན། འདི་ ནམ་ཡང་ CPU ལུ་ ལོག་མི་འགྱོ། ཐབས་འཕྲུལ་འདི་གིས་ Merkle ལོ་མ་དང་ འོག་རིམ་ཕ་མའི་ བསྡུས་རྟགས་བཟོ་ནི་ ཕྱོགས་བསྡོམས་སྦེ་ ལག་བསྟར་འབདཝ་ཨིན། ཡིག་བསྒྱུར་དང་ གོ་རིམ་ཅན་གྱི་ ཡར་རིམ་སྡོང་པོ་ དེ་ལས་ རྗེས་ཤུལ་རྩིས་རིག་ཚུ་ CPU གུ་ར་ སྡོདཔ་ཨིན། `poseidon_mode = "gpu"` དང་གཅིག་ཁར་ སོ་སོ་སྦེ་ཡོད་པའི་ སྔོན་ཞིབ་ཅིག་ མཐར་འཁྱོལ་བ་ཅིན་ མཐུད་མཚམས་ཀྱིས་ ཡིག་བསྒྱུར་ Poseidon བཅུད་བསྡུས་ཚུ་ཡང་ GPU གུ་ ཚན་པ་སྦེ་ ལག་བསྟར་འབདཝ་ཨིན། དེ་མེན་པ་ཅིན་ དེ་ཚུ་ CPU གུ་ རྩིས་བཏོནམ་ཨིན།

`proof_sidecar_max_bytes` གིས་ མཐུད་མཚམས་ཀྱིས་ བཞགཔ་དང་ བདེན་དཔྱད་འབད་མི་ ཨང་བཀོད་འབད་ཡོད་པའི་ ཐོན་དངོས་འདི་ ཚད་འཛིན་འབདཝ་ཨིནམ་དང་ དེའི་སྔོན་སྒྲིག་འདི་ བདེན་ཁུངས་བཟོ་མིའི་ མང་མཐའི་ བདེན་ཁུངས་བཀོད་རིས་ཨིན། `producer` རེའུ་མིག་གིས་ བདེན་ཁུངས་བཀལ་ནིའི་ འབད་རྩོལ་རེ་རེ་ ཚད་འཛིན་འབདཝ་ཨིན། དེ་ཡང་ དེའི་ བརྟག་ཞིབ་བར་མཚམས་དང་ དུས་ཚོད་ཚད་གཞི་ རང་སོའི་བསྐྱར་གཏང་རིམ་པ་ (1 ལས་ 64 ཚུན་) དེ་ལས་ བགོ་བཀྲམ་མ་འབད་བའི་ཧེ་མ་ ཞུགས་ཆོག་སྤྲོད་མི་ འབྱུང་ཁུངས་ཐོ་བཀོད་དང་ བརྗོད་དོན་ཚུ་ཨིན། `metal_queue_fanout` གིས་ 1 ལས་ 4 ཚུན་ ངོས་ལེན་འབདཝ་ཨིན།

ཐབས་ལམ་དང་ ཊེ་ལི་མི་ཊི་རི་ཁ་ཡིག་གི་ སྒྲིག་སྟངས་ཚུ་ `iroha3d` ལས་ཡང་ ཚབ་བཙུགས་ཚུགས།

```shell
iroha3d --fastpq-execution-mode gpu
iroha3d --fastpq-poseidon-mode cpu
iroha3d --fastpq-device-class apple-m4
iroha3d --fastpq-chip-family m4
iroha3d --fastpq-gpu-kind integrated
```

`iroha3d` གིས་ རིམ་སྒྲིག་ཡིག་སྣོད་ཅིག་ ལྷག་པའི་སྐབས་ ཁོར་ཡུག་འགྱུར་ཅན་ཚུ་ སྣང་མེད་བཞགཔ་ཨིན། དེ་འབདཝ་ལས་ གནས་གོང་འདི་ཚུ་ TOML ནང་ ཡང་ན་ ལྟག་གི་ གདམ་ཁ་ཚུ་གིས་ གཞི་སྒྲིག་འབད།
