# ཚོང་འབྲེལ་ཚུ་ {#transactions}

ཚོང་འབྲེལ་དེ་ བཀྲམ་སྤེལ་འབད་ཚུགས་པའི་ ནང་དོན་གནད་སྡུད་གྱི་འགན་ཁུར་འདི་ [བཀའ་རྒྱ་](./instructions.md) གི་རིམ་སྒྲིག་འབད་ནི་དང་ ཁག་འབགཔ་ལུ་འབོ་ནི་ དེ་ལས་ IVM བའི་ཊི་ཀོཌ་ ཡང་ན་ བདེན་ཁུངས་བཀལ་མི་ IVM ལག་ལེན་འབད་ནི་ཨིན། [རིག་རྩལ་ཅན་གྱི་ཞལ་འཆེས་ཚུ་](./smart-contracts.md)ལུ་བལྟ་བ་ཅིན་ ད་ལྟོའི་ཆིངས་ཡིག་གི་ལག་ལེན་གྱི་རྣམ་གཞག་འདི་ཨིན།

ཚོང་འབྲེལ་ཚུ་གིས་ གནས་སྟངས་བསྒྱུར་བཅོས་ཡང་ན་ ལག་ལེན་འཐབ་བཏུབ་པའི་ལཱ་ཚུ་འབདཝ་ཨིན། ལྷག་རྐྱངམ་ཅིག་གི་ཞིབ་དཔྱད་འདི་གིས་ མིང་རྟགས་བཀོད་ཡོད་པའི་འདྲི་དཔྱད་ཚུ་ཡང་ན་ མི་མང་ལྷག་ནིའི་མཇུག་སྣོད་ཚུ་ལག་ལེན་འཐབ་ཨིནམ་དང་ ཚོང་འབྲེལ་ཅིག་གསར་བསྐྲུན་མི་འབད།

ཁས་བླངས་འབད་ཡོད་པའི་སྡེབ་ཚན་ནང་ལུ་ ངོས་ལེན་འབད་ཡོད་པའི་ཚོང་འབྲེལ་འདི་ ལག་ལེན་འཐབ་མི་བཀག་ཆ་རྩིས་ཏེ་ ལག་ལེན་འཐབ་པའི་གྲུབ་འབྲས་དང་གཅིག་ཁར་ གསོག་འཇོག་འབདཝ་ཨིན། སྡེབ་ཚན་ནང་འཛུལ་ཞུགས་མ་འབད་བའི་ཧེ་མ་ ངོས་ལེན་མ་འབད་བའི་ཞུ་བ་ཚུ་ དཔེར་ན་ ནུས་མེད་ཡིག་ཤུབས་ ཡང་ན་ བང་རིམ་གྱིས་ ངོས་ལེན་མ་འབད་བའི་ ཚོང་འབྲེལ་ཚུ་ སྡེབ་ཚན་ནང་ གསོག་འཇོག་འབད་མི་བཏུབ།

སྲུང་སྐྱོབ་ཀྱི་དོན་ལུ་ རྒྱུ་དངོས་ཚུ་སྤེལ་བའི་དོན་ལས་ [རྣམ་རྟོག་ཅན་གྱི་ཚོང་འབྲེལ་ཚུ་](./anonymous-transactions.md) ལུ་བལྟ། མིང་མ་ཤེསཔ་གི་ཚོང་འབྲེལ་ཚུ་གིས་ མི་མང་རྩིས་ཐོ་ལས་རྩིས་ཐོ་ལུ་ བསྒྱུར་བཅོས་འབད་བ་ཚབ་སྦེ་ ཉེན་སྲུང་ཅན་གྱི་ རྒྱུ་དངོས་ཡིག་ཆའི་ལག་ལེན་དང་ ཁས་བླངས་དང་ ཆ་མེད་གཏང་ཐོ་བཀོད་ དེ་ལས་ ཤེས་ཡོན་སྟོང་པའི་རྟགས་མཚན་ཚུ་ ལག་ལེན་འཐབ་ཡོདཔ་ཨིན།

གདམ་ཁ་རྐྱབས་ཅན་གྱི་ གསལ་བའི་ལག་ལེན་གྱི་གྲུབ་འབྲས་ཚུ་གི་ཐད་ལུ་ བདེན་ཁུངས་ཀྱི་དཔང་རྟགས་ཚུ་གི་དོན་ལུ་ [FastPQ](./fastpq.md) FastPQ གིས་ རྒྱུན་ཆད་མེད་པར་ ལག་ལེན་གྱི་ལག་ལེན་ལག་ལེན་འཐབ་པའི་ཤུལ་ལས་ ལག་ལེན་གྱི་དཔང་རྟགས་ཚུ་ མཁོ་སྒྲུབ་འབད་དོ་ཡོདཔ་མ་ཚད་ རྒྱབ་སྐྱོར་འབད་མི་ གནས་སྟངས་སྤར་འགྱོ་བའི་དོན་ལུ་ ཐད་ངེས་ཅན་གི་བདེན་ཁུངས་ཀྱི་ བཀྲམ་སྤེལ་འབད་དོ་ཡོདཔ་ཨིན།

## Taira ལུ་ བརྟག་དཔྱད་རྐྱབས། {#try-it-on-taira}

བརྟག་ཞིབ་འབད་ཐངས་ཚུ་ ལག་ལེན་འཐབ་སྟེ་ ཐོ་བཀོད་རྩིས་མེད་མི་ མི་མང་གི་ Taira སྦྲག་དང་ བྱ་སྟབས་མ་བདེཝ་ཚུ་གི་གནས་སྟངས་བལྟ་ནིའི་དོན་ལུ་:

```bash
curl -fsS 'https://taira.sora.org/v1/explorer/blocks?page=1&per_page=3' \
  | jq '{pagination, blocks: [.items[] | {height, hash, transactions_total, transactions_rejected}]}'

curl -fsS 'https://taira.sora.org/v1/explorer/transactions?page=1&per_page=5' \
  | jq '{pagination, txs: [.items[] | {hash, block, status, executable}]}'
```

ཁྱོད་རའི་གློག་རིམ་གྱིས་ཧེ་མ་ལས་བཙུགས་ཡོད་པའི་ཚོང་འབྲེལ་ཅིག་རྗེས་སུ་འཇུག་ནིའི་དོན་ལུ་ ཐོ་ཡིག་ནང་ལས་ `hash` འདི་འདྲ་བཤུས་རྐྱབ་སྟེ་ འཚོལ་ཞིབ་ཁ་གསལ་གྱི་ལམ་ལུགས་འདི་ཞིབ་དཔྱད་འབད།

```bash
TX_HASH='<transaction-hash>'

curl -fsS "https://taira.sora.org/v1/explorer/transactions/$TX_HASH" \
  | jq '{hash, block, status, authority, executable}'
```

གནད་དོན་འདི་ ཀློག་རྐྱང་ཨིན། ཚོང་འབྲེལ་བཙུགས་དགོ་པ་ཅིན་ Norito ཤོག་སྒྲིལ་ཅིག་གུ་ མཚན་རྟགས་བཀོད་ཡོདཔ་དང་ སྡེབ་ཐག ID ངེས་བདེན་ གླ་ཆའི་ ཟུར་གནས་གནད་སྡུད དང་ བརྟག་དཔྱད་ཊོ་ཀེན་ཞབས་ཏོག གིས་མ་དངུལ་བཙུགས་པའི་ Taira རྩིས་ཐོ་དགོཔ་ཨིན།

དངུལ་ཕོགས་སྤྲོད་མི་དཔེ་སྟོན་ཚུ་ནང་ལུ་ Taira, མཐུད་སྦྲེལ་འབད་ནིའི་འཕྲུལ་ཆས་འདི་ [བརྟག་དཔྱད་དྲ་རྒྱ བཏོན་ཐོབ། XOR འབད་ནི་ Taira](/dz/get-started/sora-nexus-dataspaces.md#_4-get-testnet-xor-on-taira) འདི་བཟུམ་སྦེ་ `taira_faucet_claim.py`, དེ་ལས་ ལག་ལེན་པ་འདི་ འགོ་དང་པ་ར་ མི་མང་གི་བརྟག་དཔྱད་ཊོ་ཀེན་ཞབས་ཏོག་གྱི་ཐོག་ལས་ དངུལ་སྤྲོད་ནི་ཨིན།

```bash
export TAIRA_ACCOUNT_ID='<TAIRA_I105_ACCOUNT_ID>'
export TAIRA_FEE_ASSET=6TEAJqbb8oEPmLncoNiMRbLEK6tw

curl -fsS https://taira.sora.org/v1/accounts/faucet/puzzle | jq .
python3 taira_faucet_claim.py "$TAIRA_ACCOUNT_ID"

iroha --config ./taira.client.toml ledger asset get \
  --definition "$TAIRA_FEE_ASSET" \
  --account "$TAIRA_ACCOUNT_ID"
```

གལ་སྲིད་ བརྟག་དཔྱད་ཊོ་ཀེན་ཞབས་ཏོག་གྱི་མགུ་ཐོམ་ཡང་ན་ ཐོབ་བརྗོད་ལམ་ལུགས་འདི་གིས་ `502` སླར་ལོག་འབད་བ་ཅིན་ ཚོང་འབྲེལ་འདི་རང་ རྐྱེན་སེལ་མ་འབད་བའི་ཧེ་མ་ བསྒུག་སྟེ་ ལོག་སྟེ་འབད་རྩོལ་བསྐྱེད།

ཏེ་ Taira འཐུས་དངུལ་གྱི་རྒྱུ་དངོས་ཀྱི་ ཟུར་གནས་གནད་སྡུད འདི་ ཚོང་འབྲེལ བཏང་པའི་སྐབས་ མཐུད་སྦྲེལ་འབད་:

```bash
printf '{"gas_asset_id":"%s"}\n' "$TAIRA_FEE_ASSET" > taira.tx-metadata.json

iroha --config ./taira.client.toml \
  --metadata ./taira.tx-metadata.json \
  ledger transaction ping --msg "faucet-funded taira transaction"
```

སྤྱིར་བཏང་གི་བཞུར་རྒྱུན་ནི།

1. གྲ་སྒྲིག་འབད་ཞིནམ་ལས་ བཏོན་གཏང་པ་ཅིན་ `ready` འདི་རྫུས་མ་ཨིན་པ་ཅིན་ ཡང་ན་ དུམ་གྲ་ཅིག་བཀག་ཚུགས།
2. ཨེབ་གཏང་འབད་ཡོད་པའི་ལག་ལེན་འཐབ་ Swift ཡང་ན་ JVM ཤོག་སྒྲིལ་འདི་ ཚད་ལྡན་གྱི་སྣུམ་སྒྲོམ་བཟོ་ནི་དང་ བཏང་ནི་དང་ ཨེབ་ཐོར་ཡིག་ཚང་གི་གནས་གོང་དང་ལག་ལེན་གཉིས་ཆ་ར་བཞག་ནིའི་དོན་ལུ་ཨིན། ID ལས་འགུལ་དེ་ མཇུག་བསྡུཝ་ད་ ལྕགས་ཐག་གི་གནས་སྟངས་ལུ་མ་ལྷོད་ཚུན་ཚོད་
3. དགོས་མཁོ་ཅན་གྱི་སྐབས་ལུ་ བཏང་མི་ཡིག་ཚང་གི་མིང་ཐོ་བཀོད་དེ་ སེལ་འཐུ་འབད་ཞིནམ་ལས་ གནས་སྐབས་ཀྱི་ནང་འཁོད་ལུ་ མཐུད་མཚམས་ཐོག་ལས་ བཏང་མི་ཚུ་ རྐྱབ་སྟེ་ དབྱེ་ཞིབ་འབད་ཞིནམ་ལས་ བཏང་མི་འདི་ ངོས་འཛིན་མ་འབད་བའི་ཧེ་མར་ ཨེབ་གཏང་འབད་ཡོད་པའི་ ཡིག་སྣོད་ཚུ་ ཨིན་ཅི་མིན་ཅི་ བསྡུ་སྒྲིག་འབདཝ་ཨིན།
4. མཁོ་སྤྲོད་མི་འདི་ འགྲུལ་འཕྲིན་ནང་ལུ་ཡོདཔ་ད་ སྒྲིག་འཇུག་གི་ཡིག་སྣོད་ཚུ་བཟོ་སྟེ་ བཏང་ཞིནམ་ལས་ ལག་ལེན་གྱི་ཐོན་ཁུངས་ལུ་ ཞིབ་འཚོལ་མཇུག་བསྡུ་བར་འབད་འོང་།

དྲན་ཐོ་གནས་སྟངས་འདི་ ཡོངས་འབྲེལ་གྱི་མི་ཚེ་འཁོར་རིམ་བརྒྱུད་དེ་ ལོག་མ་འོང་ཚུན་ཚོད་ རྩིས་ཐོ་འདི་གིས་ འགལ་བ་ཅན་གྱི་ ཡོངས་འབྲེལ་གྱི་ བཀྲམ་སྤེལ་ཅིག་ བལྟ་མི་ཚུགས། དེ་འབདཝ་ལས་ དངུལ་ཁུག་དང་ བཀོལ་སྤྱོད་པའི་སྲིད་བྱུས་འདི་གིས་ གནས་གོང་ཚད་གཞི་དང་ དུས་ཡུན་ཚང་ནི་ ངོས་ལེན་འབད་ཡོད་པའི་ བཀྲམ་སྤེལ་འབད་མི་ ཐུབ་ཚད་ཅན་གྱི་ས་གནས་ཀྱི་གསོག་འཇོག་ དེ་ལས་ མཐུན་སྒྲིག་སྒོ་སྒྲིག་ཚུ་ བསྟར་སྤྱོད་འབད་དགོ།

འདི་ནང་ལུ་ `Grant` སྒྲིག་འཇུག་གི་ཐོག་ལས་ ཚོང་འབྲེལ་གསརཔ་བཟོ་ནིའི་དཔེ་ཅིག་ཨིན། འ་ནི་ཚོང་འབྲེལ་འདི་ནང་ Mouse གིས་ Alice ལུ་དམིགས་བསལ་གྱི་འགན་ཁུར་ (`role_id`) བྱིན་དོ་ཡོདཔ་ཨིན། ཁྱོད་ཀྱིས་ [ དཔེ་ཆ་ཡོངས་བསྡོམས་ ](./permissions.md#register-a-new-role) བལྟ་དགོ།

```rust
let grant_role = Grant::account_role(role_id, alice_id);
```
