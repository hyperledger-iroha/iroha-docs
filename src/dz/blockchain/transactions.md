# ཚོང་འབྲེལ་ཚུ་ {#transactions}

ཚོང་འབྲེལ་དེ་ བཀྲམ་སྤེལ་འབད་ཚུགས་པའི་ ཁེ་ཕན་གྱི་འགན་ཁུར་འདི་ [བཀའ་རྒྱ་](./instructions.md) གི་རིམ་སྒྲིག་འབད་ནི་དང་ ཁག་འབགཔ་ལུ་འབོ་ནི་ དེ་ལས་ IVM བའི་ཊི་ཀོཌ་ ཡང་ན་ བདེན་ཁུངས་བཀལ་མི་ IVM ལག་ལེན་འབད་ནི་ཨིན། [Smart Contracts](./smart-contracts.md)ལུ་བལྟ་བ་ཅིན་ ད་ལྟོའི་ཆིངས་ཡིག་གི་ལག་ལེན་གྱི་རྣམ་གཞག་འདི་ཨིན།

ཚོང་འབྲེལ་གྱི་གནས་སྟངས་འདི་ བསྒྱུར་བཅོས་འབད་ནི་དང་ ཡང་ན་ ལག་ལེན་འཐབ་ཚུགས་པའི་ལཱ་ཚུ་འབད་དོ་ཡོདཔ་ཨིན། ཀློག་རྐྱང་གི་བརྟག་དཔྱད་དེ་ ཐོ་བཀོད་ཅན་གྱི་དྲིས་ལན། ཡང་ན་ མི་མང་གིས་ ཀློག་ཐིག་མཇུག་བསྡུ་མི་ཚུ་ལག་ལེན་འཐབ་སྟེ་ཡོདཔ་ལས་ ཚོང་འབྲེལ་བཟོ་མི་ཚུགས།

བཀྲམ་སྤེལ་འབད་ཡོད་པའི་སྦྲག་ནང་ འཛུལ་ཞུགས་འབད་མི་ ཕྱིར་ཚོང་འདི་ ལག་ལེན་གྱི་གྲུབ་འབྲས་དང་གཅིག་ཁར་ ཐོ་བཀོད་འབདཝ་ཨིན། བཀྲམ་སྟོན་འབད་ནི་ལུ་ ཆ་མེད་བཏང་མི་ ཞུ་བ་ཚུ་ དཔེར་ན་ མཁལ་ཆ་མེད་ ཡང་ན་ བྱ་སྒོའི་གྲལ་ཐིག་གིས་ ཆ་མེད་གཏང་མི་ ཕྱིར་ཚོང་དེ་ དུམ་གྲ་ཅིག་ནང་ ཐོ་བཀོད་ཀྱི་དོན་ལུ་ ཆ་མེད་གཏང་ནི་མེདཔ་ཨིན་པས།

སྲུང་སྐྱོབ་ཀྱི་དོན་ལུ་ རྒྱུ་དངོས་ཚུ་སྤེལ་བའི་དོན་ལས་ [Anonymous Transactions](./anonymous-transactions.md) ལུ་བལྟ། མིང་མ་ཤེསཔ་གི་ཚོང་འབྲེལ་ཚུ་གིས་ མི་མང་རྩིས་ཁྲ་ལས་རྩིས་ཁྲ་ལུ་ བསྒྱུར་བཅོས་འབད་བ་ཚབ་སྦེ་ ཉེན་སྲུང་ཅན་གྱི་ རྒྱུ་དངོས་ཡིག་ཆའི་ལག་ལེན་དང་ ཁས་བླངས་དང་ ཆ་མེད་གཏང་ཐོ་བཀོད་ དེ་ལས་ ཤེས་ཡོན་སྟོང་པའི་རྟགས་མཚན་ཚུ་ ལག་ལེན་འཐབ་ཡོདཔ་ཨིན།

གདམ་ཁ་རྐྱབས་ཅན་གྱི་ གསལ་བའི་ལག་ལེན་གྱི་གྲུབ་འབྲས་ཚུ་གི་ཐད་ལུ་ བདེན་ཁུངས་ཀྱི་དཔང་རྟགས་ཚུ་གི་དོན་ལུ་ [FastPQ](./fastpq.md) FastPQ གིས་ རྒྱུན་ཆད་མེད་པར་ ལག་ལེན་གྱི་ལག་ལེན་ལག་ལེན་འཐབ་པའི་ཤུལ་ལས་ ལག་ལེན་གྱི་དཔང་རྟགས་ཚུ་ མཁོ་སྒྲུབ་འབད་དོ་ཡོདཔ་མ་ཚད་ རྒྱབ་སྐྱོར་འབད་མི་ གནས་སྟངས་སྤར་འགྱོ་བའི་དོན་ལུ་ ཐད་ངེས་ཅན་གི་བདེན་ཁུངས་ཀྱི་ བཀྲམ་སྤེལ་འབད་དོ་ཡོདཔ་ཨིན།

## Taira ལུ་ བརྟག་དཔྱད་རྐྱབས། {#try-it-on-taira}

བརྟག་ཞིབ་འབད་ཐངས་ཚུ་ ལག་ལེན་འཐབ་སྟེ་ ཐོ་བཀོད་རྩིས་མེད་མི་ མི་མང་གི་ Taira སྦྲག་དང་ བྱ་སྟབས་མ་བདེཝ་ཚུ་གི་གནས་སྟངས་བལྟ་ནིའི་དོན་ལུ་:

```bash
curl -fsS 'https://taira.sora.org/v1/explorer/blocks?page=1&per_page=3' \
  | jq '{pagination, blocks: [.items[] | {height, hash, transactions_total, transactions_rejected}]}'

curl -fsS 'https://taira.sora.org/v1/explorer/transactions?page=1&per_page=5' \
  | jq '{pagination, txs: [.items[] | {hash, block, status, executable}]}'
```

ཁྱོད་ཀྱིས་གནའ་བོའི་དུས་ལུ་ ཁྱོད་ཀྱི་ལག་ལེན་ནང་ བཏང་ཡོད་པའི་ ཕྱིར་ཚོང་གི་དོན་ལས་ ཐག་བཅད་དགོ་པ་ཅིན་ ཐོ་བཀོད་ནང་ལས་ `hash` ཨེབ་གཏང་འབད་ཞིནམ་ལས་ ཞིབ་འཚོལ་ཞིབ་འཚོལ་གྱི་ལམ་ཡིག་འདི་ དབྱེ་ཞིབ་འབད་:

```bash
TX_HASH='<transaction-hash>'

curl -fsS "https://taira.sora.org/v1/explorer/transactions/$TX_HASH" \
  | jq '{hash, block, status, authority, executable}'
```

གནད་དོན་འདི་ ཀློག་རྐྱང་ཨིན། ཚོང་འབྲེལ་བཙུགས་དགོ་པ་ཅིན་ Norito ཤོག་སྒྲིལ་ཅིག་གུ་ མཚན་རྟགས་བཀོད་ཡོདཔ་དང་ ID ལྕགས་ཀྱིསཔ་ཕྲང་སྟེ་འབད་ཡོདཔ་མ་ཚད་ དངུལ་སྤྲོད་ཀྱི་དོན་ལུ་ ཟད་འགྲོ་གི་རྩིས་ཁྲ་ Taira བྱིན་དགོཔ་ཨིན།

དངུལ་ཕོགས་སྤྲོད་མི་དཔེ་སྟོན་ཚུ་ནང་ལུ་ Taira, མཐུད་སྦྲེལ་འབད་ནིའི་འཕྲུལ་ཆས་འདི་ [Testnet བཏོན་ཐོབ། XOR འབད་ནི་ Taira](/dz/get-started/sora-nexus-dataspaces.md#_4-get-testnet-xor-on-taira) འདི་བཟུམ་སྦེ་ `taira_faucet_claim.py`, དེ་ལས་ ལག་ལེན་པ་འདི་ འགོ་དང་པ་ར་ མི་མང་གི་གློག་མེ་ལས་འགུལ་གྱི་ཐོག་ལས་ དངུལ་སྤྲོད་ནི་ཨིན།

```bash
export TAIRA_ACCOUNT_ID='<TAIRA_I105_ACCOUNT_ID>'
export TAIRA_FEE_ASSET=6TEAJqbb8oEPmLncoNiMRbLEK6tw

curl -fsS https://taira.sora.org/v1/accounts/faucet/puzzle | jq .
python3 taira_faucet_claim.py "$TAIRA_ACCOUNT_ID"

iroha --config ./taira.client.toml ledger asset get \
  --definition "$TAIRA_FEE_ASSET" \
  --account "$TAIRA_ACCOUNT_ID"
```

ཏེ་ Taira འཐུས་དངུལ་གྱི་རྒྱུ་དངོས་ཀྱི་ metadata འདི་ Transaction བཏང་པའི་སྐབས་ མཐུད་སྦྲེལ་འབད་:

|`POST /v1/offline/redeem` |མཉམ་འབྲེལ་ཐོག་ལས་ བསྡུ་ལེན་འབད་ནིའི་ལག་ལེན་ཅིག་བཙུགས་ |
|`GET /v1/offline/operations/{operation_id}` |ཡར་སེང་འབད་ནི་དང་ བསྐྱིན་འགྲུལ་ལེན་ནི་གི་ གནས་སྟངས་འདི་ ཀན་ནོག་ནང་ལུ་བལྟ་ନ୍ତୁ |

ལས་འགུལ་འདི་མ་བཙུགས་པའི་ཧེ་མར་ རྒྱུ་དངོས་ཚུ་གི་དོན་ལུ་ གྲ་སྒྲིག་ཡོད་མི་འདི་ བརྟག་དཔྱད་འབད་:

```bash
curl -fsS --get https://taira.sora.org/v1/offline/readiness \
  --data-urlencode 'asset_definition_id=<canonical_asset_definition_id>' \
  | jq '{ready, blockers, artifact_set}'
```

གྲ་སྒྲིག་འབད་ནི་འདི་གིས་ ཤོག་སྒྲིལ་འདི་ ལཱ་འབད་ཡོད་པའི་གཞུང་ལམ་ལུ་བསྡུགསཔ་ཨིན། ABI ༢༡ དང་ བདེན་ཁུངས་བཀལ་ཡོདཔ་ཨིན། V4 རིགས་བརྒྱུད་དང་ བསྡུ་སྒྲིག་ དེ་ལས་ སྐྱིན་ཚབ་ཀྱི་ཞུ་ཡིག་ཚུ་ ལག་ལེན་འཐབ་ཐོག་ལས་ ཨེབ་གཏང་འབདཝ་ཨིན། `application/x-norito` ཡིག་སྣོད་ཚུ་ བསྡུ་སྒྲིག་འབད་ཞིནམ་ལས་ བསྐྱིན་ཚབ་སླར་ལོག་འབདཝ་ཨིན། `202 Accepted` དང་གཅིག་ཁར་ `Location` ལས་འགུལ་གྱི་ཐོན་ཁུངས་ལུ་ བརྗོད་སྟོན་འབད་ཡོད་པའི་ཨེབ་རྟ་; ནང་བཙུགས་འབད་མི་ ཨེབ་རྟ་ཚུ་ ID རང་དབང་གི་ལྡེ་མིག་འདི་བཀྲམ་སྤེལ་འབདཝ་ཨིན།

སྤྱིར་བཏང་ཆུ་རྒྱུན་འདི་:

1. གྲ་སྒྲིག་འབད་ཞིནམ་ལས་ བཏོན་གཏང་པ་ཅིན་ `ready` འདི་རྫུས་མ་ཨིན་པ་ཅིན་ ཡང་ན་ དུམ་གྲ་ཅིག་བཀག་ཚུགས།
2. ཨེབ་གཏང་འབད་ཡོད་པའི་ལག་ལེན་འཐབ་ Swift ཡང་ན་ JVM ཤོག་སྒྲིལ་འདི་ ཀ་ནོ་ནི་ཀཱན་གྱི་སྣུམ་སྒྲོམ་བཟོ་ནི་དང་ བཏང་ནི་དང་ ཨེབ་ཐོར་ཡིག་ཚང་གི་གནས་གོང་དང་ལག་ལེན་གཉིས་ཆ་ར་བཞག་ནིའི་དོན་ལུ་ཨིན། ID ལས་འགུལ་དེ་ མཇུག་བསྡུཝ་ད་ ལྕགས་ཐག་གི་གནས་སྟངས་ལུ་མ་ལྷོད་ཚུན་ཚོད་
3. དགོས་མཁོ་ཅན་གྱི་སྐབས་ལུ་ བཏང་མི་ཡིག་ཚང་གི་མིང་ཐོ་བཀོད་དེ་ སེལ་འཐུ་འབད་ཞིནམ་ལས་ གནས་སྐབས་ཀྱི་ནང་འཁོད་ལུ་ འདྲན་འདྲ་ཐོག་ལས་ བཏང་མི་ཚུ་ རྐྱབ་སྟེ་ དབྱེ་ཞིབ་འབད་ཞིནམ་ལས་ བཏང་མི་འདི་ ངོས་འཛིན་མ་འབད་བའི་ཧེ་མར་ ཨེབ་གཏང་འབད་ཡོད་པའི་ ཡིག་སྣོད་ཚུ་ ཨིན་ཅི་མིན་ཅི་ བསྡུ་སྒྲིག་འབདཝ་ཨིན།
4. མཁོ་སྤྲོད་कर्ताའདི་ འགྲུལ་འཕྲིན་ནང་ལུ་ཡོདཔ་ད་ སྒྲིག་འཇུག་གི་ཡིག་སྣོད་ཚུ་བཟོ་སྟེ་ བཏང་ཞིནམ་ལས་ ལག་ལེན་གྱི་ཐོན་ཁུངས་ལུ་ ཞིབ་འཚོལ་མཇུག་བསྡུ་བར་འབད་འོང་།

འདི་འབདཝ་ལས་ བརྒྱུད་འཕྲིན་དང་ ལག་ལེན་པ་གི་ སྲིད་བྱུས་གིས་ གནས་གོང་ཚད་ཀྱི་ཐོ་བཀོད་, དུས་མཐའན་མཇུག་གི་དུས་ཡུན་, ངོས་ལེན་ཅན་གྱི་ དངུལ་ཕོགས་སྤྲོད་མི་ཚུ་, ཡུན་བརྟན་གནས་སྟངས་ཀྱི་གནས་སྟངས་ དེ་ལས་ མཐུན་རྐྱེན་གྱི་ སྒོ་སྒྲིག་ཚུ་ བཀག་ཐབས་འབད་འོང་།

འདི་ནང་ལུ་ `Grant` སྒྲིག་འཇུག་གི་ཐོག་ལས་ ཚོང་འབྲེལ་གསརཔ་བཟོ་ནིའི་དཔེ་ཅིག་ཨིན། འ་ནི་ཚོང་འབྲེལ་འདི་ནང་ Mouse གིས་ Alice ལུ་དམིགས་བསལ་གྱི་འགན་ཁུར་ (`role_id`) བྱིན་དོ་ཡོདཔ་ཨིན། ཁྱོད་ཀྱིས་ [ དཔེ་ཆ་ཡོངས་བསྡོམས་ ](./permissions.md#register-a-new-role) བལྟ་དགོ།

```rust
let grant_role = Grant::account_role(role_id, alice_id);
let grant_role_tx = TransactionBuilder::new(chain_id, mouse_id)
    .with_instructions([grant_role])
    .sign(mouse_private_key);
```
