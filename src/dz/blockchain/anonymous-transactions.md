# རྣམ་རྟོག་ཅན་གྱི་ཚོང་འབྲེལ་ཚུ་ {#anonymous-transactions}

Iroha ནང་མིང་མེད་པའི་ཚོང་འབྲེལ་ཚུ་གསང་བའི་རྒྱུ་དངོས་བཀོལ་སྤྱོད་ལས་བཟོ་བསྐྲུན་འབད་ཡོདཔ་ཨིན། མི་མང་གི་རྩིས་ཐོ་ལས་རྩིས་ཐོ་ལུ་ མི་མང་གི་དངུལ་སྤྲོད་ལེན་ཚུ་ བྲིས་ནིའི་ཚབ་ལུ་ དངུལ་ཁུག་འདི་གིས་ གནས་གོང་འདི་ བཀག་ཆ་འབད་ཡོད་པའི་ བཀག་ཆ་འབད་ཡོད་པའི་ བཀག་ཆའི་རྩིས་ཐོ་ནང་ལུ་སྤོ་བཤུད་འབད་དེ་ ཤེས་ཡོན་མེད་པའི་བདེན་ཁུངས་ཚུ་དང་གཅིག་ཁར་ དྲན་ཐོ་ཚུ་ ཟད་འགྲོ་བཏངམ་ཨིན།

མི་མང་གི་རྩིས་དེབ་ནང་ གསང་བའི་ལག་ལེན་ཅིག་བྱུང་ཡོདཔ་སྦེ་ ཐོ་བཀོད་འབད་དོ་ཡོདཔ་ཨིན། འདི་ནང་ལུ་ ཁས་བླངས་དང་ ཆ་མེད་གཏང་མི་ཚུ་ བརྟག་དཔྱད་ཧེཤ་དང་བྱུང་རྐྱེན་ཚུ་ ཐོ་བཀུད་འབད་དོ་ཡོདཔ་ཨིན་རུང་ འདི་ནང་ལུ་ ཡིག་ཚང་གི་ཇོ་བདག་དང་ མཁོ་སྤྲོད་འབད་མི་ དེ་ལས་ ཟད་འགྲོ་ཚུ་གི་དོན་ལུ་ གཏན་འཁེལ་གྱི་གནས་ཚད་མ་འབྲི་དོ། སྤྱིར་བཏང་ཚོང་འབྲེལ་གྱི་ཁེབས་ནང་ལུ་ བཏང་མི་རྩིས་དེ་ གསལ་སྟོན་འབད་ཚུགས་པས། འདི་འབདཝ་ལས་ "ཨན་ནོ་མིསི་" ཟེར་བའི་དོན་དག་འདི་ རང་བཞིན་གྱི་དྲ་ལམ་གི་གནས་ཚད་ ཡང་ན་རྩིས་ཐོའི་གནས་ཚད་ཀྱི་མིང་མ་ཤེསཔ་སྦེ་མེན་པའི་ རྒྱུ་དངོས་ཚུ་ སྤོ་བཤུད་འབད་ནི་ཨིན།

## ལག་ལེན་འཐབ་ནི་གི་ཐབས་ལམ་ {#how-to-use-them}

1. གསང་བའི་ རྒྱབ་སྐྱོར་འདི་ བདེན་ཁུངས་སྐྱེལ་འཕྲུལ་ཆས་ཀྱི་ཨེབ་རྟ་ཚུ་ནང་ལུ་ རྩ་སྒྲིག་འབད་བཅུགཔ་ཨིན། ཨེབ་རྟ་ཚུ་གིས་ བདེན་དཔྱད་འཕྲུལ་ཆས་གི་རྒྱབ་ཐུག་ལུ་ གྲོས་བསྟུན་འབད་དགོཔ་ཨིན། སྲིད་འཛིན་གྱི་ལྡེ་མིག་ལག་ལེན་འཐབ་སྟེ་ བདེན་དཔྱད་འབད་ནི་དང་ Poseidon/Pedersen ཚད་གཞི་ IDs དེ་ལས་ གསང་བ་ཅན་གྱི་ཁྲིམས་ལུགས་ཀྱི་བཟོ་བཀོད་གུ་ གྲོས་འཆམ་བཟོ་དགོ། ཨེབ་རྟ་དེ་མཐུན་མེད་མི་ གསང་བ་ཅན་གི་ཁྱད་ཆོས་ཚུ་ཡོད་མི་ དངོས་པོ་ཚུ་ ཡང་ན་ མཚམས་སྦྱོར་ཚུ་ ཆ་མེད་གཏང་དགོ།

## SDK དཔེ་སྟོན་འབདཝ་ཨིན། {#sdk-example}

```rust
use iroha_data_model::{
    isi::zk::{Unshield, ZkTransfer},
    prelude::{AccountId, AssetDefinitionId, InstructionBox},
    proof::{ProofAttachment, ProofBox, VerifyingKeyId},
};

fn transfer_instruction(
    asset: AssetDefinitionId,
    input_nullifier: [u8; 32],
    output_commitment: [u8; 32],
    anchor_root: [u8; 32],
    proof_bytes: Vec<u8>,
) -> InstructionBox {
    let backend = "halo2/ipa".into();
    let proof = ProofBox::new(backend, proof_bytes);
    let vk = VerifyingKeyId::new("halo2/ipa", "vk_transfer");
    let attachment = ProofAttachment::new_ref("halo2/ipa".into(), proof, vk);

    ZkTransfer::new(
        asset,
        vec![input_nullifier],
        vec![output_commitment],
        attachment,
        Some(anchor_root),
    )
    .into()
}

fn unshield_instruction(
    asset: AssetDefinitionId,
    recipient: AccountId,
    amount: u128,
    input_nullifier: [u8; 32],
    anchor_root: [u8; 32],
    proof_bytes: Vec<u8>,
) -> InstructionBox {
    let backend = "halo2/ipa".into();
    let proof = ProofBox::new(backend, proof_bytes);
    let vk = VerifyingKeyId::new("halo2/ipa", "vk_unshield");
    let attachment = ProofAttachment::new_ref("halo2/ipa".into(), proof, vk);

    Unshield::new(
        asset,
        recipient,
        amount,
        vec![input_nullifier],
        attachment,
        Some(anchor_root),
    )
    .into()
}
```

5. `CancelAnonymousAssetEscrow` དངུལ་འབྲེལ་མཐུན་རྐྱེན་དེ་བཙོང་མི་གི་ཐོན་སྐྱེད་མཐུན་རྐྱེན་ལུ་ བསྐྱལ་དོ་ཡོདཔ་ད་ སྐྱིན་འགྲུལ་གྱི་རྟགས་མ་བཀོད་མི་ལུ་བརྟེན་ཨིན།
6. `OpenAnonymousEscrowDispute` དང་ `ResolveAnonymousEscrowDispute` གིས་ རྩོད་གཞི་ཅན་གྱི་ཨེསི་ཀོརོ་ཚུ་ སྒྲུབ་བྱེད་ཀིརིཔ་ཊོ་གཱ་ར་ཕིག་ཧ་ཤི་དང་ ཐབས་ཤེས་ཚད་འཛིན་འབད་མི་ བགོ་བཤའ་ཚུ་དང་གཅིག་ཁར་ འཛིན་སྐྱོང་འཐབ་ཨིན།

[དྲི་བ་](/dz/reference/queries.md#escrow-and-proof-records)ནང་ལུ་བཀོད་ཡོད་པའི་ གསང་བའི་བར་གཏོགས་བདག་ཉརའི་ཞུ་ཡིག་ཚུ་ ལག་ལེན་འཐབ་སྟེ་ བར་གཏོགས་བདག་ཉརའི་ཐོ་ཡིག་དང་ གནས་སྟངས་ཚུ་ བརྟག་ཞིབ་འབད་ཚུགས།

## རྩིས་རིག་པ་ {#math}

འོག་གི་མཚོན་རྟགས་འདི་གིས་ གསང་བའི་རྒྱུ་དངོས་རྒྱུན་འགྲུལ་འདི་འགྲེལ་བཤད་རྐྱབ་ཨིན། ལག་ལེན་འཐབ་མི་ཚུ་གིས་ རྒྱུ་དངོས་སྲིད་བྱུས་དང་ བདེན་དཔྱད་འབད་མི་ཐོ་བཀོད་ལས་ ཤུགས་ལྡན་གློག་ལམ་དང་ ཚད་གཞི་ཨའི་ཌི་ཚུ་ལག་ལེན་འཐབ་ཨིནམ་ལས་ མཁོ་མངགས་འབད་མི་ཚུ་གིས་ ཁས་བླངས་དང་ ཆ་མེད་གཏང་མི་ དེ་ལས་ བདེན་ཁུངས་བཱའིཊི་ཚུ་ དངུལ་ཁུག་/བདེན་དཔྱད་འབད་མི་གི་ མཐོང་མ་ཚུགས་པའི་ ཐོན་འབྲས་སྦེ་ བརྩི་དགོཔ་ཨིན།

བཀག་སྡོམ་འབད་ཡོད་པའི་དྲན་ཐོ་འདི་ འདི་བཟུམ་སྦེ་འགྲེལ་བཤད་རྐྱབ་ཚུགས།

$$
n = (\mathsf{asset}, \mathsf{amount}, \mathsf{owner}, \rho)
$$

འདི་ནང་ `owner` འདི་ ཐོབ་མི་གི་བལྟ་བཤལ་ཡང་ན་ ཟད་འགྲོ་བཏང་མི་རྒྱུ་ཆ་ལས་ འབྱུང་ཡོདཔ་དང་ `rho` འདི་ དྲན་ཐོ་གང་བྱུང་ཅིག་ཨིན།

དྲན་ཐོའི་ཁས་ལེན་འདི་གསང་བའི་ཁས་ལེན་ཅིག་ཨིན།

$$
C = \mathsf{Commit}(\mathsf{asset}, \mathsf{amount}, \mathsf{owner}, \rho)
$$

ད་ལྟོའི་གསང་བ་གི་གནས་སྤེལ་གྱི་འཁོར་ལམ་ཚུ་གི་དོན་ལུ་ མི་མང་གི་ནང་འཁོད་ནང་ལུ་ ཡིག་ཚང་གི་འགན་ཁུར་ཚུ་དང་ ཆ་མེད་གཏང་མི་ཚུ་ དེ་ལས་ Merkle རྩ་བ དང་ རྒྱུ་དངོས རྟགས དེ་ལས་ སྡེབ་ཐག རྟགས ཚུ་ཡོདཔ་ཨིན། ས་མཚམས་འདི་ ཁས་ལེན འབྲེལ་བ་བཟུམ་ཅིག་བཟོ་ཡོདཔ་ཨིན།

$$
C = H_c(\mathsf{amount}, \rho, \mathsf{owner\_tag}, \mathsf{asset\_tag})
$$

དྲན་ཐོ་ཅིག་ཟད་འགྲོ་བཏང་པའི་སྐབས་ དངུལ་ཁུག་འདི་གིས་ ནུས་མེད་བཟོ་མི་ཅིག་ཐོནམ་ཨིན།

$$
N = H_n(\mathsf{spend\_key}, \rho, \mathsf{asset\_tag}, \mathsf{chain\_tag})
$$

`N` ནི་མི་མང་ཡིན། འདི་གིས་ དྲན་འཛིན་འདི་གསལ་སྟོན་མི་འབད་ དེ་འབདཝ་ད་ དྲན་འཛིན་དང་རིམ་སྒྲིག་དེ་གི་དོན་ལུ་ བརྟན་ཏོག་ཏོ་ཡོདཔ་ལས་ Iroha གིས་ ནུས་མེད་བཟོ་མི་གཅིག་དང་གཅིག་ཁར་ ཟད་འགྲོ་གཉིས་པ་བཏང་མི་འདི་ བཀག་ཆ་འབད་ཚུགས།

ཁས་བླངས་ཤིང་གིས་ དྲན་ཐོ་ཡོད་པའི་བདེན་དཔང་འབདཝ་ཨིན། གལ་སྲིད་ དངུལ་ཁུག་ཅིག་གིས་ ཁས་བླངས་ `C_i` ཟད་འགྲོ་བཏང་པ་ཅིན་ བདེན་ཁུངས་འདི་ནང་ `C_i` ལས་ འཕྲལ་གྱི་མི་མང་རྩ་བ་ལུ་ སྒེར་གྱི་མར་ཀལ་འགྲུལ་ལམ་ཅིག་ཚུདཔ་ཨིན།

$$
\mathsf{MerkleRoot}(C_i, \mathsf{path}) = R
$$

བཀག་ཆ་ལས་བཀག་ཆ་ལུ་སྤོ་བཤུད་འབད་ནིའི་དོན་ལུ་ བདེན་ཁུངས་འདི་གིས་ གནས་གོང་ཉམས་སྲུང་ཡང་བཀག་དམ་འབདཝ་ཨིན།

$$
\sum \mathsf{inputs} = \sum \mathsf{outputs}
$$

བཀག་ཆ་མེད་པའི་དོན་ལུ་ མི་མང་གི་དངུལ་འབོར་འདི་ཚུདཔ་ཨིན།

$$
\sum \mathsf{inputs} = \mathsf{public\_amount} + \sum \mathsf{private\_change}
$$

ཕུལ་བའི་བདེན་དཔང་འདི་ བཅུད་བསྡུས་འབད་ཚུགས།

$$
\mathsf{Verify}(\mathsf{vk}, \mathsf{public\_inputs}, \pi) = \mathsf{true}
$$

འདི་ནང་ `public_inputs` འདི་ ཁས་བླངས་དང་ ཆ་མེད་གཏང་མི་ རྩ་བ་ རྒྱུ་དངོས་ངོ་རྟགས་ རིམ་སྒྲིག་ངོ་རྟགས་ དེ་ལས་ མི་མང་སྲུང་སྐྱོབ་མ་འབད་བའི་ དངུལ་འབོར་གང་རུང་ཚུ་ཨིན། དཔང་པོ་འདི་ནང་ དྲན་ཐོའི་དངུལ་འབོར་དང་ གང་བྱུང་ ཟད་འགྲོ་བཏང་བའི་རྒྱུ་ཆ་ དེ་ལས་ མར་ཀལ་འགྲུལ་ལམ་ཚུ་ཡོདཔ་ཨིན། བདེན་དཔྱད་འབད་མི་ཚུ་གིས་ བདེན་ཁུངས་བདེན་དཔྱད་འབད་ཞིནམ་ལས་ ཐོན་འབྲས་ཁས་བླངས་ཚུ་ ཁ་སྐོང་བརྐྱབ་སྟེ་ ཨིན་པུཊི་ནུ་ལི་ཕའི་ཡར་ཚུ་ ཟད་འགྲོ་བཏང་ཡོདཔ་བཟུམ་སྦེ་ རྟགས་བཀལ་ཏེ་ ལེ་ཇར་གནས་སྟངས་འདི་ འགྱུར་བཅོས་འབདཝ་ཨིན།

## མི་མང་གིས་ག་ཅི་མཐོང་ནི་ཨིན་ན? {#what-is-public}

མིང་མ་ཤེསཔ་གི་ཚོང་འབྲེལ་ཚུ་གིས་ བརྟག་ཞིབ་འབད་ཚུགས་མི་ གནད་དོན་ཆ་མཉམ་ལུ་ གསང་བ་བཟོཝ་མེན། ད་ལྟོའི་བར་ན་ཡང་ གནད་སྡུད་འདི་ མི་མང་གིས་མཐོང་ཚུགསཔ་ཨིན།

- ཚོང་འབྲེལ་གྱི་ཧ་ཤི་དང་ སྡེབ་ཚན་གྱི་མཐོ་ཚད་ དེ་ལས་ གོ་རིམ་སྒྲིག་ནི།
- ཞུ་ཡིག་འདི་གིས་ སྒེར་གྱི་འཛུལ་སྒོ་ཡང་ན་ བསྐྱར་ལོག་བཀོད་རིས་ལག་ལེན་མ་འཐབ་ཚུན་ཚོད་ བཙུགས་མི་ཚོང་འབྲེལ་དབང་འཛིན་འདི།
- རྒྱུ་དངོས་ངེས་ཚིག་ལག་ལེན་འཐབ་བཞིན་པ།
- ནུས་མེད་བཟོ་མི་དང་ ཐོན་འབྲས་ཁས་ལེན་ཚུ།
- བདེན་དཔང་ཧ་ཤི་ཚུ་དང་ བདེན་དཔྱད་-ལྡེ་མིག་གཞི་བསྟུན་ཚུ་ དེ་ལས་ གདམ་ཁ་ཅན་གྱི་ཡིག་ཤུབས་ཧ་ཤི་ཚུ་
- `Unshield` གི་དོན་ལུ་ མི་མང་དངུལ་འབོར་དང་ ཐོབ་མི་རྩིས་ཐོ།
- མིང་མེད་པའི་ཨེས་ཀོ་རོ་བཙོང་མི་དང་ཉོ་མི་ གནས་རིམ་ དུས་ཚོད་ཀྱི་རྟགས་མཚན་ དེ་ལས་ སྒྲུབ་བྱེད་ཧ་ཤི་ཚུ་ཨིན།

བཟོ་བཀོད་གློག་རིམ་ཚུ་ དེ་འབདཝ་ལས་ མི་མང་མེ་ཊ་ཌེ་ཊ་འདི་གིས་ ཁྱོད་ཀྱིས་ཉེན་སྲུང་འབད་ནི་ལུ་ འབད་རྩོལ་བསྐྱེད་མི་ ཚོང་འབྲེལ་མཐུན་ལམ་འདི་ གསལ་སྟོན་མི་འབད།
