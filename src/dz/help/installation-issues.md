# གཞི་བཙུགས་ཀྱི་དཀའ་ངལ་སེལ་ཐབས། {#troubleshooting-installation-issues}

འ་ནི་ཆ་ཤས་འདི་ནང་ལུ་ Iroha 3 གཞི་བཙུགས་ཀྱི་དོན་ལུ་ དཀའ་ངལ་སེལ་ཐབས་ཀྱི་བསླབ་བྱ་ཚུ་ གྲོས་འདེབས་འབདཝ་ཨིན། ཁྱོད་ཀྱིས་མྱོང་དོ་ཡོད་པའི་དཀའ་ངལ་འདི་ ནཱ་ལུ་མ་བཤད་པ་ཅིན་ ང་བཅས་དང་གཅིག་ཁར་ [Telegram](https://t.me/hyperledgeriroha) ཟེར་མི་ཁ་ཐུག་ལས་འབྲེལ་བ་འཐབ་གནང་།

## བརྟག་ཞིབ་མགྱོགས་དྲགས་ {#quick-checks}

མཐུད་སྦྲེལ་གྱི་འཛོལ་བ་མང་ཤོས་ཅིག་གིས་ ས་གནས་བཞི་ལས་གཅིག་ལས་བྱུང་དོ་ཡོདཔ་ཨིན།

- Rust ལག་ཆས་ཀྱི་ཐོ་བཀོད་འདི་ ལས་རིམ་གོང་འཕེལ་གྱི་ལཱ་གི་ས་ཁོངས་ནང་བཙུགས་མི་ འདྲ་བཤུས་ལས་རྙིང་ཤོས་ཨིན།
- `cargo` ཡང་ན་ `rustc` མཐུན་རྐྱེན་གཞན་ཅིག་ལུ་ སེལ་འཐུ་འབད། `rustup`
- ལམ་ལུགས་བཟོ་སྐྲུན་གྱི་ལག་ཆས་ཚུ་ མེདཔ་ཐལ་དོ་ཡོདཔ་ཨིན། དཔེར་ན་ C compiler, `pkg-config` ཡང་ན་ CMake
- གཞི་རྟེན་འགྱུར་བཅོས་ཚུ་ བསྒྱུར་བཅོས་འབད་བའི་ཤུལ་ལུ་ ཨེབ་གཏང་འབད་ཡོད་མི་ ཡང་ན་ ས་གནས་ཀྱི་བཟོ་སྐྲུན་གྱི་ལག་ཆས་ཚུ་

Iroha ལས་ཁུངས་ཀྱི་དངུལ་ཁང་ནང་ལས་འགོ་བཙུགས་ཏེ་:

```bash
rustup show
cargo --version
rustc --version
cargo metadata --no-deps
```

གལ་སྲིད་ `cargo metadata` ཕྱིར་འབུད་འབད་མ་ཐོབ། ཁྱོད་ཀྱིས་ `pnpm refresh:iroha --source /path/to/iroha` བརྩོན་ཤུགས་མ་བསྐྱེད་པའི་ཧེ་མ་ ས་གནས་ཀྱི་ལག་ཆས་བཅུད་སྒྲོམ་འདི་ སེལ་འཐུ་འབད། ག་ཅི་སྨོ་ཟེར་བ་ཅིན་ ད་ལྟོའི་ data-model schema བཟོ་ནིའི་དོན་ལུ་ refresh གིས་ Kagami ལུ་འབོ་ཚུགས།

### བརྟག་དཔྱད་འབད་ Rust འདྲ་ཕབ། {#check-rust-version}

ཁྱོད་ཀྱིས་ `cargo` གི་བདེན་པའི་བཟོ་བཀོད་དང་ `rustc`གི་བདེན་པའི་བཟོ་བཅོས་ཚུ་ཡོད་མི་འདི་ ལེགས་ཤོམ་སྦེ་ལག་ལེན་འཐབ་དགོ། ད་ལྟོའི་ཡར་ཐུག་ལུ་ལཱ་འབད་སའི་ས་སྒོ་འདི་ `rust-version = "1.92"` གསལ་སྟོན་འབད་ཞིནམ་ལས་ ལག་ཆས་སྤེལ་ལམ་དེ་ `rust-toolchain.toml` ལུ་སྦྲགས་གཏང་ཨིན།

```bash
$ cargo -V
$ cargo 1.93.1 (...)
```

དེ་ལས་ཤུལ་ལས་

```bash
$ rustc --version
$ rustc 1.93.1 (...)
```

ཁྱོད་ཀྱིས་ ཚད་མཐོ་ཤོས་ཡོད་པ་ཅིན་ ཁྱོད་དྲག་ཆེ། ཁྱོད་ཀྱིས་ ཚད་དམའ་བ་ཡོད་པ་ཅིན་ འདི་བཟོ་བཅོས་འབད་ནིའི་དོན་ལུ་ བཀའ་རྒྱ་འདི་ལག་ལེན་འཐབ་བཏུབ་ཨིན།

```bash
$ rustup toolchain update stable
```

### གཞི་བཙུགས་ས་སྒོ་ཚུ་ བརྟག་དཔྱད་འབད། {#check-installation-location}

ཁྱོད་ཀྱིས་འགྱུར་ལྡེ་མིག་གི་ཨང་གྲངས་ཉུང་སུ་ཅིག་ཐོབ་སྟེ་ ལག་ཆས་ཀྱི་ཐིག་ཁྲམ་འདི་ ད་ལྟོའི་བར་དུ་བཟོ་བཅོས་འབད་ཞིནམ་ལས་ ལཱ་འབད་མ་ཚུགས་པ་ཅིན་ ... དེ་སྦེ་སླབ་ནི་དེ་ སྤྱིར་བཏང་དཀའ་ངལ་ཅིག་ཨིན་ དེ་འབདཝ་ད་ དེ་གི་དོན་ལུ་ ཐབས་ལམ་གཅིག་ཡང་མེད་ཟེར་བརྗོད་དགོ།

དང་པ་ར་ ཁྱོད་ཀྱིས་ ལག་ལེན་འཐབ་དགོ་པའི་ ཝིར་སི་འདི་ ག་ཏེ་བཙུགས་ཡོདཔ་ཨིན་ན་ ངེས་གཏན་འབད་དགོ།

```bash
$ rustup which rustc
$ rustup which cargo
```

ལག་ཆས་ཐིག་ཁྲམ་ཚུ་གི་ལག་ལེན་འཐབ་མི་བཙུགས་ནི་ཚུ་ སྤྱིར་བཏང་ལུ་ `~/.rustup/toolchains/stable-*/bin/`ནང་ལུ་ཨིན། འདི་ཨིན་པ་ཅིན་ ཁྱོད་ཀྱིས་

```bash
$ rustup toolchain update stable
```

འདི་གིས་ ཁྱོད་ཀྱི་དཀའ་ངལ་ཚུ་ སེལ་ཚུགས་འོང་།

### གཞི་སྒྲིག་གི་ Rust འདྲ་བཤུས་ཚུ་ བརྟག་དཔྱད་འབད་ {#check-the-default-rust-version}

གདམ་ཁ་གཞན་ཅིག་འདི་ ཁྱོད་ཀྱིས་ `stable` ལག་ཆས་བཅུད་ཀྱི་ལྡེ་མིག་འདི་ ད་ལྟོའི་བར་ན་ཡང་ གཞི་སྒྲིག་འབད་མ་ཚུགསཔ་ཨིན།

```bash
$ rustup default stable
```

ཁྱོད་ཀྱིས་ `nightly` འདྲ་བཤུས་ཅིག་བཙུགས་ཏེ་ཡོད་ ཡང་ན་ Rust འདྲ་བཤུད་ཅིག་ གཞི་སྒྲིག་འབད་ཡོད་རུང་ འདི་བཏོན་མ་བཏུབ་པ་ཅིན་ འདི་འབྱུང་ཚུགས།

### Rust འདྲ་བཤུས་གཞན་ཚུ་ཡོད་མེད་ བརྟག་དཔྱད་འབད་ {#check-if-there-are-other-rust-versions}

དཀའ་ངལ་སེལ་ཐབས་ཀྱི་ལྡེ་མིག་འདི་ འཕྲོ་མཐུད་འབད་ཞིནམ་ལས་ ང་བཅས་ཀྱིས་ shell aliases འཐོབ་ཚུགས་ནི།

```bash
$ type rustc
$ type cargo
```

`rustup which *` བཀྲམ་སྤེལ་འབད་བའི་སྐབས་ ཁྱོད་ཀྱིས་མཐོང་མི་ ས་གནས་གཞན་ཚུ་ལུ་བཏོན་པ་ཅིན་ ཁྱོད་ལུ་དཀའ་ངལ་ཅིག་ཡོདཔ་ཨིན། འདི་བཟུམ་མའི་ མིང་མིང་བཙུགས་ནི་དེ་ ལེ་ཤ་མིན་འདུག་ཟེར་གཟིགས།

```bash
$ alias rustc "~/.rustup/toolchains/stable-*/bin/rustc"
$ alias cargo "~/.rustup/toolchains/stable-*/bin/cargo"
```

ནང་འཁོད་གི་ མནོ་བསམ་འདི་ མ་གཞི་ ཁྱོད་ཀྱིས་ shell aliases འདི་ག་དེ་སྦེ་སྒྲིག་རུང་ བསྒྱུར་བཅོས་འབད་ཚུགས།

ཐབས་ལམ་འཇམ་ཏོང་ཏོ་འདི་ ཁྱོད་ཀྱིས་ལག་ལེན་འཐབ་མ་བཏུབ་པའི་ འདྲ་བཤུས་ཚུ་བཏོན་གཏང་ནི་ཨིན།

ཨིན་རུང་ འདི་འབད་ནི་ལས་སླབ་ནི་འཇམ་ཏོང་ཏོ་ཅིག་ཨིན་ ག་ཅི་སྨོ་ཟེར་བ་ཅིན་ rustup གི་ འདྲ་བཤུས་ཚུ་ བསྡུ་སྒྲིག་འབད་ཞིནམ་ལས་ ཁྱོད་རའི་ལག་པར་བཙུགས་ཏེ་ཡོདཔ་ཨིན། སྤྱིར་བཏང་ལུ་ ༢ རྐྱངམ་གཅིག་ཡོད་: ལམ་ལུགས་པེ་ཅེཊ་འཛིན་སྐྱོང་འཕྲུལ་ཆས་དང་ ཌའི་ལོག་གི་གནས་སྟངས་ནང་ གཞི་བཙུགས་འབད་ཡོདཔ་ཨིན། འ་ནི་ལྷབ་སྦྱང་འདི་གི་ འགོ་བཙུགས་ནང་ལུ་ ཁྱོད་ཀྱིས་བཀའ་འདི་ལག་ལེན་འཐབ་པའི་སྐབས་ལུ་ ཁྱོད་ཀྱི་ home folder ཟེར་འབད། དང་པ་གི་དོན་ལས་ ཁྱོད་ཀྱི་ (Linux) ཁྱབ་སྤེལ་ལག་དེབ་ (`apt remove rust`) ཚུ་བལྟ་དགོ། འ་ནི་མཇུག་གི་དོན་ལུ་ ཨེབ་གཏང་འབད།

```bash
$ rustup toolchain list
```

དེ་ལས་ `<toolchain>` གི་དོན་ལས་ (སྒོ་ཟུར་གྱི་ལྡེ་མིག་ཚུ་མེད་པར་) ཨིན།

```bash
$ rustup remove <toolchain>
```

དེ་གི་ཤུལ་ལས་ འདི་ལུ་བརྟག་དཔྱད་འབད་

```bash
$ cargo --help
```

བཀའ་རྒྱ་མ་ཐོབ་པའི་འཛོལ་བ་ཐོན་ཡོདཔ་ཨིན། འདི་འབདཝ་ལས་ ཁྱོད་ཀྱིས་ Rust ཤུགས་ཅན་གྱི་ལག་ཆས་ལྡེ་མིག་ installedམེདཔ་ཨིན། དེ་ལས་ run:

```bash
$ rustup toolchain install stable
```

## དཀའ་ངལ་སེལ་ཐབས། Python ལག་ཆས་སྤེལ་ཐིག་ {#troubleshooting-python-toolchain}

ཁྱོད་ཀྱིས་ Python Wheel Package འདི་ [Python client setup](/dz/guide/tutorials/python.md) གི་སྐབས་ལུ་ pip ལག་ལེན་འཐབ་ཐོག་ལས་བཙུགས་པའི་སྐབས་ ཁྱོད་ཀྱིས་ "iroha_python-*.whl འདི་ འ་ནི་གློག་བརྙན་ནང་ལུ་ རྒྱབ་སྐྱོར་འབད་མི་ འཁོར་ལམ་ཅིག་མེན་"བཟུམ་གྱི་འཛོལ་བ་མཐོང་ཚུགས།

འ་ནི་འཛོལ་བ་འདི་གིས་ pip འདི་དུས་ཡུན་མ་ཆོགཔ་འབདཝ་ལས་ ཁྱོད་ཀྱིས་འདི་ ད་ལྟོའི་གནས་སྟངས་ལུ་ བསྐྱར་ཞིབ་འབད་དགོཔ་ཨིན། དང་པ་རང་ ཁྱོད་ཀྱི་ OS ལུ་ ད་ལྟོའི་ གནས་སྟངས་ཚུ་འཚོལ་ནིའི་དོན་ལུ་ བསྐྱར་ཞིབ་འབད་ནི་དང་ལམ་ལུགས་གོང་འཕེལ་གཏང་ནི་དེ་ གྲོས་ཐག་ཆོད་འབདཝ་ཨིན།

ག་དེམ་ཅིག་སྦེ་འདི་ལཱ་མ་འབད་བ་ཅིན་ ཁྱོད་ཀྱིས་ `pip` ཌའི་ཀར་ལོ་མི་ཊར་ཚུ་གི་དོན་ལུ་ ད་ལྟོའི་བཟོ་བཀོད་འབད་ཚུགས་འོང་།

`python -m pip install --upgrade pip`

འདི་གི་དོན་ལུ་ བརྟག་ཞིབ་འབད་ `pip` ཁྱོད་ཀྱི་ཁྱིམ་གྱི་ཐོ་ཡིག་ནང་ལུ་ གཞི་བཙུགས་འབད་ཡོདཔ་ཨིན། འདི་གི་དོན་ལུ་ ཁྱོད་ཀྱིས་ Run `whereis pip` བརྟག་དཔྱད་འབད་ `/home/username/.local/bin/pip` ཐབས་ལམ་ཚུ་གི་གྲལ་ཁར་ཨིན། ག་དེམ་ཅིག་སྦེ་ དེ་མེན་པ་ཅིན་ ཁྱོད་ཀྱི་ཤེལ་འདི་ ད་ལྟོའི་བར་ན་ཡང་ སེལ་འཐུ་འབད། `PATH` བསྒྱུར་བཅོས་འབད་ཚུགས།

གནད་དོན་འདི་ འཕྲོ་མཐུད་དེ་ར་ཡོད་པ་ཅིན་ ཁྱོད་ཀྱིས་ [ ལུ་ ང་བཅས་དང་ ](/dz/help/) འབྲེལ་བ་འཐབ་སྟེ་ གྲུབ་འབྲས་ཚུ་ སྙན་ཞུ་འབད་དགོ།

```
python --version
python3 --version
pip --version
pip3 --version
```
