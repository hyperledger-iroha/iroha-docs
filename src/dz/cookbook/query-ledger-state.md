# གནད་དོན་ཁག་གི་ཐོ་ཡིག་ {#query-ledger-state}

## གྲུབ་འབྲས་ {#outcome}

Taira JSON ཐོན་ཁུངས་ཚུ་ ལྷག་སྟེ་ དགོ་པའི་ས་སྒོ་ཚུ་ གདམ་འཐུ་འབད་ཞིནམ་ལས་ CLI དང་ Rust SDK ལས་ Torii བསྡུ་ཚོགས་དྲི་དཔྱད་ཚུ་ ཚགས་མ་དང་ གོ་རིམ་སྒྲིག་ནི་ ཞབས་ཏོག་སྤྲོད་མིའི་ཕྱོགས་ཀྱི་ ས་སྒོ་གདམ་འཐུ་ ཤོག་ལེབ་ཚད་བཀག་ དེ་ལས་ ནང་དོན་མི་མངོན་པའི་ `next_cursor` འཕྲོ་མཐུད་དང་གཅིག་ཁར་ ལག་ལེན་འཐབ། ཁྱོད་ཀྱིས་ ཐོ་ཡིག་བཏོན་ནི་ལུ་ མིང་རྟགས་བཀོད་ཡོད་པའི་ དབྱེ་བ་ཅན་གྱི་དྲི་དཔྱད་ཚུ་ ལག་ལེན་མི་འཐབ། དེ་ཚུ་གིས་ ཐོ་ཡིག་ཚུ་ ཚགས་མ་ གོ་རིམ་ ཡང་ན་ ས་སྒོ་གདམ་འཐུ་ མི་འབད།

## དགོས་མཁོ་ཚུ་ {#prerequisites}

- `curl`, `jq`, Node.js 24,དང་ ད་ལྟོ་ `iroha` CLI འདི་ཚུ་ཨིན།
- ཀློག་ཐངས་རྐྱངམ་ཅིག་ Taira ཐོབ་ཚུགསཔ་ཨིན།
- CLI དཔེ་ཚུ་གི་དོན་ལུ་ Taira ཡང་ན་ བཟོ་བཏོན་འབད་ཡོད་པའི་ ས་གནས་དྲ་རྒྱ་ཅིག་གི་ ཞབས་ཏོག་ལེན་མིའི་རིམ་སྒྲིག་ཅིག།
- Rust གི་དཔེ་མཚོན་ནང་ལུ་ ལས་འགུལ་ཅིག་གིས་ དམིགས་གཏད་ཐོ་བཀོད་འབད་ཡོད་པའི་དྲ་ལམ་དང་འདྲཝ་སྦེ་ Iroha གི་འབྱུང་ཁུངས་བསྐྱར་བཅོས་འབད་ཡོདཔ་ཨིན།

## རིམ་པ་ཚུ་ {#steps}

### Taira ཐོན་ཁུངས་ནང་ལས་ ཤོག་ལེབ་ཅིག་བཏོན་གནང་། {#_1-page-through-a-public-taira-resource}

ཐོན་ཁུངས་ལམ་ཚུ་ ཌེཤ་བོརཌ་དང་ ཐ་མག་བརྟག་དཔྱད་ཀྱི་དོན་ལུ་ ཕན་ཐོགས་ཅན་ཨིན། JSON འདྲི་ཞིནམ་ལས་ ཤོག་ལེབ་རེ་རེ་ལུ་བསྡམ་བཞག་ཞིནམ་ལས་ ལན་ཞིབ་དཔྱད་འབད་བའི་ཤུལ་ལས་ གློག་རིམ་ལུ་དགོ་པའི་ས་སྒོ་ཚུ་རྐྱངམ་ཅིག་ལས་འགུལ་འབད།

::: code-group

```bash [curl]
curl -fsS -H 'Accept: application/json' --get \
  https://taira.sora.org/v1/domains \
  --data-urlencode 'sort=id' \
  --data-urlencode 'limit=5' \
  --data-urlencode 'include_total=true' \
  | jq '{total, next_cursor, ids: [.items[].id]}'
```

```js [Node.js]
const root = 'https://taira.sora.org'
const seen = new Set()
let cursor = null

do {
  const url = new URL('/v1/domains', root)
  url.search = new URLSearchParams({
    sort: 'id',
    limit: '5',
    ...(cursor ? { cursor } : {}),
  })

  const response = await fetch(url, {
    headers: { Accept: 'application/json' },
  })
  if (!response.ok)
    throw new Error(`Taira returned HTTP ${response.status}`)

  const page = await response.json()
  for (const domain of page.items) {
    if (seen.has(domain.id)) throw new Error(`duplicate ${domain.id}`)
    seen.add(domain.id)
    console.log(domain.id)
  }
  cursor = page.next_cursor
} while (cursor)
```

:::

HTTP ཁ་ཐོག་འདི་གིས་ `limit` དང་ ནང་དོན་མི་མངོན་པའི་ `next_cursor` གིས་ ཤོག་ལེབ་སོ་སོར་ ཕྱེཝ་ཨིན། དེ་ `null` མ་འགྱུར་ཚུན་ མ་བསྒྱུར་བར་ `cursor` སྦེ་ ལོག་གཏང་། `total` འདི་ ཁྱོད་ཀྱིས་ `include_total=true` ཞུ་བའི་སྐབས་རྐྱངམ་ཅིག་ འབྱུངམ་ཨིན། དེ་ལུ་ ཐོ་ཡིག་ཆ་ཚང་ བཤེར་ནི་ དགོཔ་ཨིན།

### 2. CLI གིས་ བསྡུ་ཚོགས་ཅིག་ ཚགས་མ་བཙུགས {#_2-filter-a-collection-with-the-cli}

CLI གིས་ ཚགས་མ་དང་ ས་སྒོ་གདམ་འཐུ་ དེ་ལས་ ཤོག་ལེབ་ཀྱི་ཚད་ཚུ་ བསྡུ་ཚོགས་ཀྱི་ མཐའ་མཇུག་ས་ཚིགས་ལུ་ གཏང་ཞིནམ་ལས་ ཤོག་ལེབ་གཅིག་ `{items, next_cursor}` གསལ་སྟོནམ་ཨིན། `next_cursor` འདི་ `null` མ་འགྱུར་ཚུན་ ཤུལ་བཟུང་ནི་ལུ་ `--all` ཁ་སྐོང་བརྐྱབ།

```bash
DOMAIN_FILTER='id = "universal.universal"'

iroha --config ./localnet/client.toml \
  ledger domain list \
  --filter "$DOMAIN_FILTER" \
  --select id \
  --limit 1
```

ཚགས་མ་བཙུགས་ནི་འདི་ ཤོག་ལེབ་སོ་སོར་ཕྱེ་བའི་ཧེ་མ་ འབྱུངམ་ཨིན། བསྡུ་ཚོགས་རེ་རེ་གིས་ རང་གི་ས་སྒོ་ཚུ་རྐྱངམ་ཅིག་ ངོས་ལེན་འབདཝ་ཨིན། མ་ཤེས་པའི་ས་སྒོ་ཅིག་གིས་ ག་ནི་ཡང་ མཐུན་སྒྲིག་མ་འབད་ནིའི་ཚབ་ལུ་ `invalid_filter` ལོག་བྱིནམ་ཨིན།

### 3. གནས་བརྟན་གནས་བརྡའི་ལྡེ་མིག་ཐོག་ལས་དབྱེ་ཞིབ་འབད་ནི་ {#_3-sort-by-a-stable-metadata-key}

བསྡུ་ཚོགས་ཀྱི་ གོ་རིམ་ལྡེ་མིག་ཚུ་གིས་ གསལ་བསྒྲགས་འབད་ཡོད་པའི་ ས་སྒོ་ཚུ་དང་ `metadata.<key>` ངོས་ལེན་འབདཝ་ཨིན། མར་འབབ་གོ་རིམ་གྱི་དོན་ལུ་ ལྡེ་མིག་གི་ཧེ་མ་ `-` བཙུགས། ལྡེ་མིག་མེད་པའི་ གྲལ་ཐིག་ཚུ་ ཡར་འཕར་གོ་རིམ་ནང་ དང་པ་ལུ་ འོངམ་ཨིན། གནས་གོང་འདྲ་མཉམ་ཡོད་པ་ཚུ་ གྲལ་ཐིག་གི་ངོ་རྟགས་ལུ་ བརྟེན་ཏེ་ གོ་རིམ་སྒྲིགཔ་ཨིན། དེ་འབདཝ་ལས་ བསྡུ་ཚོགས་ཆ་མཉམ་ནང་ རྒྱུན་མཐུད་སྦེ་ ཡོད་པའི་ ལྡེ་མིག་ཅིག་ ལག་ལེན་འཐབ།

```bash
iroha --config ./localnet/client.toml \
  --machine \
  account list \
  --sort metadata.key \
  --limit 10 \
  | jq '[.items[] | {id, metadata}]'
```

`--select` འདི་ ཞབས་ཏོག་སྤྲོད་མིའི་ཕྱོགས་ཀྱི་ ས་སྒོ་གདམ་འཐུ་ (projection) ཅིག་ཨིན། `--select id,metadata` གིས་ ས་སྒོ་དེ་ཚུ་རྐྱངམ་ཅིག་ ལོག་བྱིནམ་ཨིནམ་དང་ མ་ཤེས་པའི་ས་སྒོ་ཅིག་གིས་ `invalid_select` ལོག་བྱིནམ་ཨིན། མིང་རྟགས་བཀོད་ཡོད་པའི་ དབྱེ་བ་ཅན་གྱི་དྲི་དཔྱད་ཚུ་གིས་ ཨ་རྟག་རང་ རྣམ་གྲངས་ཆ་ཚང་ ལོག་བྱིནམ་ལས་ དེ་ཚུ་གི་གྲུབ་འབྲས་ཚུ་ གོང་ལུ་བཟུམ་སྦེ་ `jq` ཡང་ན་ JavaScript གིས་ ཞབས་ཏོག་ལེན་མིའི་ཕྱོགས་ལུ་ ས་སྒོ་གདམ་འཐུ་འབད།

### 4. Rust བསྐྱར་རྒྱུག་གིས་གསང་བའི་འདྲེན་རྟགས་ཚུ་རྟིང་བདའ་ནི་ {#_4-let-the-rust-iterator-follow-opaque-cursors}

`limit` གིས་ ཤོག་ལེབ་ཀྱི་ཚད་ གཞི་སྒྲིག་འབདཝ་ཨིན། `list` གིས་ ལོག་བྱིན་མི་ བསྐྱར་རྒྱུག་པ་འདི་གིས་ ད་ལྟོའི་ཤོག་ལེབ་ ལག་ལེན་འཐབ་ཚར་བའི་ཤུལ་ལས་རྐྱངམ་ཅིག་ ཞབས་ཏོག་སྤྲོད་མིའི་ `next_cursor` གིས་ ཤོག་ལེབ་ཤུལ་མ་ ཞུཝ་ཨིན། `take` གིས་ གཏན་ཚིག་ཅན་གྱི་གྲུབ་འབྲས་ ཚད་བཀག་འབདཝ་ཨིན། ནཱ་ལུ་ `client` འདི་ `iroha::blocking::Client` ཅིག་ཨིན།

```rust
use iroha::collections::{Collection, ListQuery, SortKey};

let query = ListQuery::new().sort_by(SortKey::asc("id")).limit(5);

for definition in client.list(Collection::AssetDefinitions, query).take(25) {
    let definition = definition?;
    println!("{:?} {:?}", definition.get("id"), definition.get("name"));
}
```

`next_cursor` འདི་ དེ་གི་བསྡུ་ཚོགས་ འགྲུལ་ལམ་ ཚགས་མ་ དེ་ལས་ གོ་རིམ་ཚུ་ལུ་ བཅིངས་ཏེ་ཡོད་པའི་ ནང་དོན་མི་མངོན་པའི་ ལྡེ་མིག་ཚན་ (keyset) གྱི་ གནས་ས་ཅིག་ཨིན། དེ་ ནམ་ཡང་ དབྱེ་དཔྱད་ ཡང་ན་ བཟོ་བསྐྲུན་མ་འབད། ཚགས་མ་ ཡང་ན་ གོ་རིམ་སོ་སོ་ཅིག་དང་གཅིག་ཁར་ ལོག་སྟེ་ ལག་ལེན་མ་འཐབ། Torii གིས་ དེ་ `invalid_cursor` གིས་ བཀག་ཆ་འབདཝ་ཨིན། ཤོག་ལེབ་ཚུ་གི་བར་ན་ བཙུགས་མི་ ཡང་ན་ བཏོན་གཏང་མི་ གྲལ་ཐིག་ཚུ་གིས་ ཤུལ་མའི་ཤོག་ལེབ་ཅིག་ནང་ འབྲེལ་བ་མེད་པའི་ གྲལ་ཐིག་ཚུ་ ལོག་སྟོན་ནི་ ཡང་ན་ མ་སྟོན་པར་བཞག་ནི་ ནམ་ཡང་ མི་འབྱུང་།

## བརྟག་དཔྱད་འབད་ {#verify}

མངའ་ཁོངས་ཚགས་མ་ངོ་མ་འདི་ `universal.universal` རྐྱངམ་ཅིག་སླར་ལོག་འབད་དགོ། མཐར་འཁྱོལ་ཅན་གྱི་ CLI ཕྱིར་ཐོན་རྐྱངམ་ཅིག་རྩིས་རྐྱབ་ནི་ལས་ གྲུབ་འབྲས་བདེན་དཔྱད་འབད།

```bash
iroha --config ./localnet/client.toml \
  --machine \
  ledger domain list --filter "$DOMAIN_FILTER" --select id --limit 1 \
  | jq -e '[.items[].id] == ["universal.universal"]'
```

ཤོག་ལེབ་སོ་སོར་ཕྱེ་མི་ གློག་རིམ་དྲི་དཔྱད་ཚུ་གི་དོན་ལུ་ ཤོག་ལེབ་ཚུ་ནང་ ID ཚུ་ ལོག་སྟེ་ མ་འབྱུངམ་དང་ ཞུ་ཡོད་པའི་ གཏན་ཚིག་ཅན་གྱི་ཚད་བཀག་ ནམ་ཡང་ མ་བརྒལཝ་ དེ་ལས་ བཀག་ཆ་འབད་ཡོད་པའི་ འོད་རྟགས་ཅིག་ ཡིག་ཐོག་བཀོད་ཡོད་པའི་ ཞིབ་དཔྱད་ས་ཚིགས་ལས་ ལོག་འགོ་བཙུགསཔ་ཚུ་ཡང་ བརྟག་དཔྱད་འབད།

## དཀའ་ངལ་སེལ་ཐབས། {#troubleshooting}

- ཨང་གཅིག་གི་དྲི་བཀོད་འདི་ བསྐྱར་ཞིབ་འབད་ཚུགས་མི་ ཕུལ་ཊར་, དབྱེ་རིམ་སྒྲིག་འབད་ནི་, ཤོག་ལེབ་བཟོ་ནི་ ཡང་ན་ ཕབ་ལེན་འབད་བཏུབ་པའི་ གནས་ཚད་ཚུ་ཆ་མ་བཞགཔ་ཨིན། འདི་ཚུ་གི་དོན་ལུ་ དགོས་མཁོ་ཡོད་པའི་སྐབས་ལུ་ འོས་འབབ་ཅན་གྱི་ཐོ་ཡིག་གི་དྲི་བཀོད་ལག་ལེན་འཐབ་དགོ།
- `limit` འདི་ ཤོག་ལེབ་ཀྱི་ཚད་ཨིན་ གྲུབ་འབྲས་ཡོངས་བསྡོམས་མེན། དེ་ ཉུང་ཤོས་ 1 དང་ མཐོ་ཤོས་ མཐུད་མཚམས་ཀྱི་ `torii.app_api_max_list_limit` (སྔོན་སྒྲིག་ 500) ཨིན། Torii གི་སྔོན་སྒྲིག་འདི་ 100 ཨིན། ཧེང་བཀལ་ལྷག་ནི་ལུ་ `next_cursor`, `--all`, ཡང་ན་ SDK བསྐྱར་རྒྱུག་པ་ ཤུལ་བཟུང་།
- `invalid_cursor` ཟེར་མི་འདི་གིས་ འོད་རྟགས་འདི་ བཟོ་རྣམ་མ་ཚང་བ་ ཡང་ན་ བསྡུ་ཚོགས་ འགྲུལ་ལམ་ ཚགས་མ་ ཡང་ན་ གོ་རིམ་སོ་སོ་ཅིག་ལུ་ གཏོགསཔ་སྦེ་ སྟོནམ་ཨིན། དྲི་དཔྱད་འདི་ ལོག་འགོ་བཙུགས། ནང་དོན་མི་མངོན་པའི་ གནས་གོང་འདི་ བཅོ་ཁ་རྐྱབ་ནི་ལུ་ འབད་རྩོལ་མ་བསྐྱེད།
- གདམ་ཁ་དམ་དྲག་ཅན་གྱི་ ཚགས་མ་ཅིག་གིས་ `next_cursor` `null` མེན་པའི་ ཤོག་ལེབ་ཐུང་ཀུ་ ཡང་ན་ སྟོངམ་ཅིག་ ལོག་བྱིན་ཚུགས། དེ་ `null` མ་འགྱུར་ཚུན་ ཤུལ་བཟུང་སྟེ་ འཕྲོ་མཐུད་འབད།
- `--filter` དང་ HTTP `filter` ཚད་བཟུང་གིས་ `id = "universal.universal"` ཡང་ན་ `exists(metadata.tier)` བཟུམ་གྱི་ Torii གི་ ཡིག་ཐོག་བརྡ་སྤྲོད་ ལེནམ་ཨིན། JSON ཆ་རྐྱེན་བརྗོད་པའི་དངོས་པོ་ (predicate object) ཅིག་ ཚགས་མའི་ཡིག་ཐོག་ མེནམ་ལས་ ནུས་མེད་ `filter` སྦེ་ འཐུས་ཤོར་འབྱུངམ་ཨིན།
- ཚད་བཀག་མེད་པའི་ ཁྱབ་ཚད་རྒྱ་ཆེ་བའི་ དྲི་དཔྱད་ཚུ་གིས་ མཉམ་རོགས་ཀྱི་ལཱ་དང་ ཞབས་ཏོག་ལེན་མིའི་ དྲན་ཚད་ ཡར་སེང་འབདཝ་ཨིན། ལག་ལེན་འཐབ་མི་ལུ་ འོས་འབབ་ཅན་གྱི་ གཏན་ཚིག་ཅན་གྱི་ཚད་བཀག་དང་ ཤོག་ལེབ་ཀྱི་ཚད་ཅིག་ གཞི་སྒྲིག་འབད།
- མིང་རྟགས་བཀོད་ཡོད་པའི་ `/v1/query` འདི་ ཐོ་ཡིག་བཏོན་ནིའི་ API མེན། དེ་གིས་ རྐྱང་པའི་དྲི་དཔྱད་ཚུ་དང་ ཚགས་མ་མེད་པའི་ བསྐྱར་རྒྱུག་འགོ་བཙུགས་ ཉུང་ཤས་ཅིག་ ངོས་ལེན་འབདཝ་ཨིན། `FindDomains` ཡང་ན་ `FindAssetDefinitions` བཟུམ་གྱི་ གཞན་ཚུ་ `signed_query_shape_not_admitted` གིས་ བཀག་ཆ་འབདཝ་ཨིན། བསྡུ་ཚོགས་ཀྱི་ མཐའ་མཇུག་ས་ཚིགས་ཚུ་དང་ CLI `list` བཀའ་རྒྱ་ཚུ་ དེ་ལས་ SDK བསྡུ་ཚོགས་ལྷག་མི་ཚུ་གིས་ ཚགས་མ་ གོ་རིམ་ དེ་ལས་ འོད་རྟགས་ཀྱི་ བརྡ་སྤྲོད་ གཅིག་པ་ མཉམ་སྤྱོད་འབདཝ་ཨིན།
