# གནད་དོན་ཁག་གི་ཐོ་ཡིག་ {#query-ledger-state}

## གྲུབ་འབྲས་ {#outcome}

ལྷག་ཞིནམ་ལས་ ལས་འགུལ་ Taira JSON ཐོན་ཁུངས་ཚུ་ དེ་ལས་ ཡིག་དཔར་རྐྱབ་ཡོད་པའི་ Iroha འདྲི་དཔྱད་ཚུ་ ཚགས་མ་དང་ གཏན་ཚིག་ཅན་གྱི་ཤོག་ལེབ་བཟོ་ནི་ དབྱེ་སེལ་འབད་ནི་ ལེན་ནིའི་ཚད་ དེ་ལས་ གདོང་ཕྱོགས་རྐྱངམ་ཅིག་ འོད་རྟགས་འཕྲོ་མཐུད་དང་གཅིག་ཁར་ལག་ལེན་འཐབ། ཁྱོད་ཀྱིས་ སར་བར་གྱིས་ གདོང་བསྐྱོད་འབད་ཡོད་པའི་ `--select` ཊུ་པལ་འདི་ བརྟག་ཞིབ་མ་འབད་བའི་ཧེ་མ་ སེལ་འཐུ་འབད་མི་པར་བརྙན་ལུ་ བློ་གཏད་ནི་ལས་ འཛེམ་འོང་།

## དགོས་མཁོ་ཚུ་ {#prerequisites}

- `curl`, `jq`, Node.js 24,དང་ ད་ལྟོ་ `iroha` CLI འདི་ཚུ་ཨིན།
- ཀློག་ཐངས་རྐྱངམ་ཅིག་ Taira ཐོབ་ཚུགསཔ་ཨིན།
- ཐོ་བཀོད་ཅན་གྱི་དྲི་བ་དཔེར་ན་ Taira གི་དོན་ལུ་ ཞབས་ཏོག་ལེན་མི རིམ་སྒྲིག ཡང་ན་ ས་གནས དྲ་རྒྱ བཟོ་སྐྲུན་འབད་ཡོད ཀྱི་དོན་ལུ་
- Rust གི་དཔེ་མཚོན་ནང་ལུ་ ལས་འགུལ་ཅིག་གིས་ དམིགས་གཏད་ཐོ་བཀོད་འབད་ཡོད་པའི་དྲ་ལམ་དང་འདྲཝ་སྦེ་ Iroha གི་འབྱུང་ཁུངས་བསྐྱར་བཅོས་འབད་ཡོདཔ་ཨིན།

## རིམ་པ་ཚུ་ {#steps}

### Taira ཐོན་ཁུངས་ནང་ལས་ ཤོག་ལེབ་ཅིག་བཏོན་གནང་། {#_1-page-through-a-public-taira-resource}

ཐོན་ཁུངས་ལམ་ཚུ་ ཌེཤ་བོརཌ་དང་ ཐ་མག་བརྟག་དཔྱད་ཀྱི་དོན་ལུ་ ཕན་ཐོགས་ཅན་ཨིན། JSON འདྲི་ཞིནམ་ལས་ ཤོག་ལེབ་རེ་རེ་ལུ་བསྡམ་བཞག་ཞིནམ་ལས་ ལན་ཞིབ་དཔྱད་འབད་བའི་ཤུལ་ལས་ གློག་རིམ་ལུ་དགོ་པའི་ས་སྒོ་ཚུ་རྐྱངམ་ཅིག་ལས་འགུལ་འབད།

::: code-group

```bash [curl]
curl -fsS -H 'Accept: application/json' --get \
  https://taira.sora.org/v1/domains \
  --data-urlencode 'sort=id:asc' \
  --data-urlencode 'limit=5' \
  --data-urlencode 'offset=0' \
  --data-urlencode 'count_mode=exact' \
  | jq '{total, ids: [.items[].id]}'
```

```js [Node.js]
const root = 'https://taira.sora.org'
const limit = 5
const seen = new Set()

for (let offset = 0; ; offset += limit) {
  const url = new URL('/v1/domains', root)
  url.search = new URLSearchParams({
    sort: 'id:asc',
    limit: String(limit),
    offset: String(offset),
    count_mode: 'exact',
  })

  const response = await fetch(url, {
    headers: { Accept: 'application/json' },
  })
  if (!response.ok)
    throw new Error(`Taira returned HTTP ${response.status}`)

  const page = await response.json()
  for (const domain of page.items) {
    if (seen.has(domain.id)) throw new Error(`duplicate ${domain.id}`)
    seen.add(domain.id)
    console.log(domain.id)
  }
  if (page.items.length < limit || seen.size >= page.total) break
}
```

:::

HTTP མཐུད་ངོས འདི་གིས་ `limit` དང་ `offset` ལག་ལེན་འཐབ། རྒྱུན་ལམ དེ་གིས་རིན་གོང་དམའ་བའི གྲངས་བརྩི ཐབས་ལམ ལག་ལེན་འཐབ་པའི་སྐབས་ `total` མ་བཀོད་མི་ ཡང་ན་ ཚད་བཀག་ཡོད་མི་དེ་སྤྱིར་བཏང་ཨིནམ་སྦེ་བརྩི།

### CLI ཐོ་བཀོད་འབད་ཡོད་པའི་དྲི་བ་འདི་སེལ་འཐུ་དང་ བཏོན་གཏང་། {#_2-filter-and-batch-a-typed-cli-query}

CLI གིས་ ཐོ་བཀོད་འབད་བཏུབ་པའི་ འདྲི་དཔྱད་འདི་རིམ་སྒྲིག་སྦེ་བཟོ་སྟེ་ ཞབས་ཏོག་གི་མུ་མཐུད་ཚད་འཛིན་ཚུ་ནང་འཁོད་ལུ་ ལྟ་རྟོག་འབདཝ་ཨིན། འདི་ནང་ལུ་ མནོ་བསམ་གྱི་གྲུབ་འབྲས་དེ་ གྲལ་ཐིག་གཅིག་ལས་བརྒལ་མེད་དོ་ཡོདཔ་ད་ `--fetch-size 1` གིས་ ཕར་དང་ཚུར་ འགྲོ་འགྲུལ་ཐེངས་རེ་ལུ་ བསྡུ་ལེན་འབད་མི་ ཨང་ཆེ་ཤོས་ཀྱི་ཨང་གྲངས་ཚུ་ བཏོན་དོ་ཡོདཔ་ཨིན།

```bash
DOMAIN_PREDICATE='{"equals":[{"field":"id","value":"wonderland.universal"}]}'

iroha --config ./localnet/client.toml \
  ledger domain list filter "$DOMAIN_PREDICATE" \
  --limit 1 \
  --offset 0 \
  --fetch-size 1
```

ཤོག་ལེབ་མ་བཟོ་བའི་ཧེ་མ་ ཚགས་མ་འབྱུངམ་ཨིན། འདྲི་དཔྱད་དམིགས་བསལ་ཡིག་དཔར་རྐྱབས་ཡོད་པའི་སྔོན་དཔག་ཚུ་ལག་ལེན་འཐབ། རྩིས་ཐོ་ཡང་ན་རྒྱུ་དངོས་ཅིག་གི་དོན་ལུ་ སྔོན་དཔག་ཅིག་ མངའ་ཁོངས་ཅིག་གི་དོན་ལུ་ ཉེན་སྲུང་དང་ལྡནམ་སྦེ་ ལོག་སྟེ་ལག་ལེན་འཐབ་མི་བཏུབ།

### 3. གནས་བརྟན་གནས་བརྡའི་ལྡེ་མིག་ཐོག་ལས་དབྱེ་ཞིབ་འབད་ནི་ {#_3-sort-by-a-stable-metadata-key}

ཡིག་དཔར་རྐྱབས་ཡོད་པའི་འདྲི་དཔྱད་དབྱེ་སེལ་འདི་ མེ་ཊ་ཌེ་ཊ་ལྡེ་མིག་གཅིག་གུ་ཚིག་མཛོད་རིག་པ་ཨིན། ལྡེ་མིག་དེ་མེད་པའི་རྣམ་གྲངས་ཚུ་གིས་ མཉེན་ཆས་ལག་བསྟར་མཉེན་ཆས་གྱིས་ངེས་འཛིན་འབད་ཡོད་པའི་གོ་རིམ་འདི་རྗེས་སུ་འཇུགཔ་ཨིན།

```bash
iroha --config ./localnet/client.toml \
  --machine \
  ledger account list all \
  --verbose \
  --sort-by-metadata-key key \
  --order asc \
  --limit 10 \
  --offset 0 \
  --fetch-size 2 \
  | jq '[.[] | {id, metadata}]'
```

ཐོ་བཀོད་འབད་ཡོད་པའི་ CLI གིས་ `--select` JSON དབྱེ་ཞིབ་འབད་དེ་ སེལ་བྱེད་གནས་གོང་སྡེ་ཕར་བཏངམ་ཨིན། ཨིན་རུང་ ད་ལྟོའི་ལྗིད་ཡང་དྲི་བཀོད་ DSL གིས་ ཞབས་ཏོག་སྤྲོད་མིའི་ཕྱོགས་ལས་སེལ་བྱེད་འདི་དབྱེ་ཞིབ་མི་འབད། ད་ལྟོ་འདི་གུ་ཕྱིར་སྟོན་གྱི་ཁས་ལེན་མ་བཟོ། དམིགས་གཏད་ལག་བསྟར་མཉེན་ཆས་ཀྱིས་རྒྱབ་སྐྱོར་འབད་ཚར་ཞིནམ་ལས་རྐྱངམ་གཅིག་ དབྱེ་བ་ཅན་གྱི་ SDK ཕྱིར་སྟོན་ལག་ལེན་འཐབ། ཡང་ན་ གོང་གི་བཟུམ་སྦེ་ `jq` ཡང་ན་ JavaScript ཐོག་ལས་ བདེན་དཔྱད་གྲུབ་འབྲས་ཞབས་ཏོག་ལེན་མིའི་ཕྱོགས་ལས་ཕྱིར་སྟོན་འབད།

### 4. Rust བསྐྱར་རྒྱུག་གིས་གསང་བའི་འདྲེན་རྟགས་ཚུ་རྟིང་བདའ་ནི་ {#_4-let-the-rust-iterator-follow-opaque-cursors}

`Pagination` གིས་ གཏན་ཚིག་ཅན་གྱི་གྲུབ་འབྲས་ཆ་ཚན་འདི་མཚམས་འཇོག་འབདཝ་ཨིན། `FetchSize` གིས་ སར་བར་ཆ་ཚན་རེ་རེ་ཚད་འཛིན་འབདཝ་ཨིན། སླར་ལོག་འབད་ཡོད་པའི་བསྐྱར་ལོག་འདི་གིས་ སར་བར་གྱིས་བཟོ་བཏོན་འབད་ཡོད་པའི་འོད་རྟགས་ལག་ལེན་འཐབ་ཐོག་ལས་ འཕྲོ་མཐུད་ཞུ་བ་ཚུ་ དྭངས་གསལ་ཅན་སྦེ་གཏངམ་ཨིན།

```rust
use std::num::NonZeroU64;

use iroha::data_model::{
    prelude::FindAssetsDefinitions,
    query::{
        builder::QueryBuilderExt as _,
        parameters::{FetchSize, Pagination},
    },
};

let definitions = client
    .query(FindAssetsDefinitions::new())
    .with_pagination(Pagination::new(NonZeroU64::new(25), 0))
    .with_fetch_size(FetchSize::new(NonZeroU64::new(5)))
    .execute_all()?;

for definition in definitions {
    println!("{} {}", definition.id(), definition.name());
}
```

`ForwardCursor` འདི་དབང་འཛིན་ལུ་བཅའ་མར་གཏོགསཔ་ཨིན། བྱ་རིམ་གྱི་གནས་སྟངས་དང་ ཕར་འགྱོ་སའི་རྐྱངམ་ཅིག་ཨིན། འདི་ནམ་ཡང་ བརྟག་ཞིབ་མ་འབད། དེ་རྩ་སྒྲིག་མ་འབད། ཡང་ན་ དབང་འཛིན་ཚུ་གི་བར་ན་བགོ་བཤའ་མ་རྐྱབས། ཡང་ན་ Torii གི་ནང་ལུ་ རྒྱུན་འགྲུལ་འཐབ་ཚུགས་པའི་བསྐྱར་གསོ་འབད་ནིའི་རྟགས་མཚན་སྦེ་བཞག་ནི་མི་འོང་། གལ་སྲིད་འདི་མཇུག་བསྡུ་བ་ཅིན་ ངོ་མའི་དྲི་བ་དེ་ ཐབས་ཤེས་ཀྱིས་ལག་ལེན་གི་གནས་ཚད་ཀྱི་བརྟག་དཔྱད་སྒོ་ཁར་ ལོག་འགོ་བཙུགས་དགོ།

## བརྟག་དཔྱད་འབད་ {#verify}

མངའ་ཁོངས་ཚགས་མ་ངོ་མ་འདི་ `wonderland.universal` རྐྱངམ་ཅིག་སླར་ལོག་འབད་དགོ། མཐར་འཁྱོལ་ཅན་གྱི་ CLI ཕྱིར་ཐོན་རྐྱངམ་ཅིག་རྩིས་རྐྱབ་ནི་ལས་ གྲུབ་འབྲས་བདེན་དཔྱད་འབད།

```bash
iroha --config ./localnet/client.toml \
  --machine \
  ledger domain list filter "$DOMAIN_PREDICATE" \
  --limit 1 --offset 0 --fetch-size 1 \
  | jq -e 'length == 1 and .[0] == "wonderland.universal"'
```

ཤོག་ལེབ་ཅན་གྱི་གློག་རིམ་འདྲི་དཔྱད་ཚུ་གི་དོན་ལུ་ ཤོག་ལེབ་ཚུ་གི་བར་ན་ IDs ཚུ་བསྐྱར་ལོག་མི་འབད་བརྟག་དཔྱད་འབད་དགོཔ་ཨིན་ ཞུ་བ་འབད་ཡོད་པའི་གཏན་ཚིག་ཅན་གྱི་ཚད་གཞི་འདི་ནམ་ཡང་མ་བརྒལ་ཏེ་ ཡིག་ཆ་བཟོ་ཡོད་པའི་ཞིབ་དཔྱད་ས་ཚིགས་ལས་ དུས་ཚོད་རྫོགས་པའི་འོད་རྟགས་འདི་ལོག་སྟེ་འགོ་བཙུགསཔ་ཨིན།

## དཀའ་ངལ་སེལ་ཐབས། {#troubleshooting}

- ཨང་གཅིག་གི་དྲི་བཀོད་འདི་ བསྐྱར་ཞིབ་འབད་ཚུགས་མི་ ཕུལ་ཊར་, དབྱེ་རིམ་སྒྲིག་འབད་ནི་, ཤོག་ལེབ་བཟོ་ནི་ ཡང་ན་ ཕབ་ལེན་འབད་བཏུབ་པའི་ གནས་ཚད་ཚུ་ཆ་མ་བཞགཔ་ཨིན། འདི་ཚུ་གི་དོན་ལུ་ དགོས་མཁོ་ཡོད་པའི་སྐབས་ལུ་ འོས་འབབ་ཅན་གྱི་ཐོ་ཡིག་གི་དྲི་བཀོད་ལག་ལེན་འཐབ་དགོ།
- `fetch_size` འདི་ ཀླད་ཀོར་མེན་པའི་ བེཆ་བརྡ་སྟོན་ཨིན་ གྲུབ་འབྲས་ཡོངས་བསྡོམས་ཚད་གཞི་མེན། ད་ལྟོའི་སྔོན་སྒྲིག་འདི་ `100` ཨིན་ དེ་ལས་ རན་ཊའིམ་འདི་གིས་ དེ་གི་མཐོ་ཤོས་ལས་ལྷག་པའི་གནས་གོང་ཚུ་ ངོས་ལེན་མི་འབད།
- མ་ཤེས་མི་ དུས་ཚོད་རྫོགས་མི་ ཡང་ན་ ཕྱི་རྒྱལ་གྱི་འོད་རྟགས་འདི་ བསམ་ཞིབ་འབད་དེ་ ལོག་སྟེ་ལག་ལེན་འཐབ་མི་བཏུབ། འདྲི་དཔྱད་འདི་ལོག་འགོ་བཙུགས། དྭངས་གསལ་མེད་པའི་གནས་གོང་འདི་ཉམས་བཅོས་འབད་ནི་ལུ་དཔའ་བཅམ་མ་བཏུབ།
- མེ་ཊ་ཌེ་ཊ་དབྱེ་སེལ་འདི་སྤྱིར་བཏང་ས་སྒོ་དབྱེ་སེལ་མེན། རྣམ་གྲངས་རེ་རེ་གིས་ སེལ་འཐུ་འབད་ཡོད་པའི་ལྡེ་མིག་འདི་འབག་མ་ཚུགས་པ་ཅིན་ བརླག་སྟོར་ཤོར་བའི་ལྡེ་མིག་གོ་རིམ་འདི་ཡིག་ཆ་བཟོཝ་ཨིན་ ཡང་ན་ ཐབས་ཤེས་གཞན་ཅིག་གདམ་ཁ་རྐྱབས།
- CLI གིས་ `--select` བརྟག་ཞིབ་འབད་ཞིནམ་ལས་ ཕར་འཕུལ་དོ་ཡོདཔ་ཨིན་རུང་ ད་ལྟོའི་ ཞབས་ཏོག་གིས་ འོད་ཐུང་སེལ་འཐུ་འབད་ནིའི་ ཐུབ་པ་དེ་ བསྐྱར་ཞིབ་མ་འབད་བར་ བཞག་དོ་ཡོདཔ་ཨིན། སེལ་འཐུ་འབད་ནིའི་དུས་ཚོད་ལུ་ ཌའི་ལོག་གི་རྒྱབ་སྐྱོར་མ་ཐོབ་པ་ཅིན་ མགྲོན་པོ་གི་ཁ་ཐུག་ལས་ གྲུབ་འབྲས་བཏོན་ནི་ལག་ལེན་འཐབ་དགོ།
- ཚད་མེད་པའི་འདྲི་དཔྱད་རྒྱ་ཆེ་བ་ཚུ་གིས་ མཉམ་རོགས་ཀྱི་ལཱ་དང་ མཁོ་སྤྲོད་འབད་མི་དྲན་ཚད་ དེ་ལས་ འོད་རྟགས་ཀྱི་ཚེ་སྲོག་ཉེན་ཁ་ཡར་སེང་འབདཝ་ཨིན། གཏན་ཚིག་ཅན་གྱི་ཚད་གཞི་དང་ ཉོ་སྤྱོད་པ་ལུ་འོས་འབབ་ཡོད་པའི་ ཕེཆ་ཚད་ཅིག་གཞི་སྒྲིག་འབད།
- མི་མང་གི་ JSON ཐོན་ཁུངས་ཀྱི་བརྡ་དོན་དང་ ཐོ་བཀོད་ཅན་གྱི་ ཐོ་བཀོད་ཀྱི་བརྡ་དོན་ཚུ་ འབྲེལ་བ་ཡོདཔ་ཨིན་རུང་ བསྒྱུར་བཅོས་འབད་ཚུགས་པའི་ བརྡ་བཀོད་བཟོ་རྣམ་མེདཔ། ཐོ་བཀུད་ཅན་གྱི་བརྡ་དོན་ནང་ SDK ཡང་ན་ CLI གདམ་ཁ་རྐྱབ་ཨིན།
